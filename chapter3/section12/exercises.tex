\begin{enumerate} [label=\textbf{\arabic*.}, leftmargin=0em]

  \item Let $Y$ be a scheme of finite type over an algebraically closed field $k$. Show that the function
        \begin{equation*}
          \varphi(y) = \dim_k(\goth{m}_y/\goth{m}_y^2)
        \end{equation*}
        is upper semicontinuous on the set of closed points of $Y$.

        \begin{proof}
          Apply (12.7.2) to the sheaf of relative differentials $\Omega_{Y/k}$.
          For each closed point $y \in Y$, we have a natural isomorphism (II, 8.7A)
          \begin{equation*}
            \gm_y/\gm_y^2 \cong \Omega_{Y / k} \otimes k(y).
          \end{equation*}
        \end{proof}

  \item Let $\{X_t\}$ be a family of hypersurfaces of the same degree in $\P_k^n$. Show that for each $i$, the function $h^i(X_t, \fO_{X_t})$ is a constant function of $t$.

        \begin{proof}
          Hypersurfaces of same degree in projective space $\P^n$ have the same cohomology.
          For any hypersurface $X \subset \P^n$ of degree $d$, we have an exact sequence
          \[ \begin{tikzcd}
              0 \arrow[r] & \fO_{\P^n}(-d) \arrow[r] & \fO_{\P^n} \arrow[r] & \fO_X \arrow[r] & 0,
            \end{tikzcd} \]
          and $\cdh^i(X, \fO_X)$ is independent of the choice of $X$.
        \end{proof}

  \item[\textbf{4.}] Let $Y$ be an integral scheme of finite type over an algebraically closed field $k$. Let $f : X \to Y$ be a flat projective morphism whose fibers are all integral schemes. Let $\fL, \fM$ be invertible sheaves on $X$, and assume for each $y \in Y$ that $\fL_y \cong \fM_y$ on the fiber $X_y$. Then show that there is an invertible sheaf $\fN$ on $Y$ such that $\fL \cong \fM \otimes f^* \fN$.

        \begin{proof}
          If $f$ is flat, then $\fL \otimes \fM^{-1}$ is flat over $Y$.
          For each $y \in Y$, $(\fL \otimes \fM^{-1})_y \cong \fO_{X_y}$ on the fibre $X_y$, and each $X_y$ is integral and projective over $k(y)$, so we have $$\cdh^0(X_y, (\fL \otimes \fM^{-1})_y) = \cdh^0(X_y, \fO_{X_y}) = 1.$$
          Thus, $f_*(\fL \otimes \fM^{-1})$ is locally free of rank one on $Y$ (12.9), say $f^*f_*(\fL \otimes \fM^{-1}) = \fN$ for some $\fN \in \pic{Y}$.
          Applying $f^*$ to both sides, and using the fact that the natural morphism
          \begin{equation*}
            f^*f_*(\fL \otimes \fM^{-1}) \to \fL \otimes \fM^{-1}
          \end{equation*}
          is actually an isomorphism because $f_*(\fL \otimes \fM^{-1})$ is locally free of rank one, we obtain our desired result.
        \end{proof}

  \item[\textbf{5.}] Let $Y$ be an integral scheme of finite type over an algebraically closed field $k$. Let $\mathscr{E}$ be a locally free sheaf on $Y$, and let $X = \P(\fE)$. Then show that $\pic{X} \cong (\pic{Y}) \times \Z$. This strengthens (II, Ex. 7.9).

        \begin{proof}
Let $r$ be the rank of $\fE$, $\fO_X(1)$ the canonical invertible sheaf of $X$, and $\pi : X \to Y$ the natural projection morphism.
For any point $y \in Y$, let $k(y)$ be the residue field of $y$, and let $\P_y$ be the fiber of $\pi$ over $y$, which is a projective $r$-space over $k(y)$.
There is an obvious map
\begin{align*}
\varphi : \pic{Y} \times \Z & \to \pic{X}                      \\
(\fL, d)                    & \mapsto \fO_X(d) \otimes \pi^*\fL.
\end{align*}
First, we show $\varphi$ is injective.
If we have an isomorphism
\begin{equation*}
\fO_X(d) \otimes \pi^*\fL \cong \fO_X(d') \otimes \pi^* \fL',
\end{equation*}
then it is clear by checking on fibres that $d = d'$, so it remains to show $\pi^* : \pic{Y} \to \pic{X}$ is injective.
We have $\pi_*\fO_X = \fO_Y$ (II, 7.11a), and for any $\fN \in \pic{Y}$, the projection formula gives
\begin{equation*}
\pi_*\pi^*\fN \cong \pi_*\fO_X \otimes \fN \cong \fN,
\end{equation*}
so if $\pi^*\fN \cong \fO_X$, then \(\fN \cong \pi_*\fO_X \cong \fO_Y.\) 
Hence, $\varphi$ is injective.

The hart part is to show $\varphi$ is surjective.
Since $\pi$ is a flat projective morphism, we can apply the results of \SS 12.
In particular, if $\fM$ is an invertible sheaf on $X$, then $\fM_y$ is an invertible sheaf on $\P_y$ for every $y \in Y$, so we can write $\fM_y \cong \fO(d_y)$ for some $d_y \in \Z$.
If we can show $d_y$ is independent of $y \in Y$, then by (Ex. 12.4) there exists an invertible sheaf $\fN$ on $Y$ such that $\fM \cong \fO_X(d) \otimes \pi^*\fN$ for some $d \in \Z$, which proves $\varphi$ is surjective.

We proceed by showing the minimum value of $d_y$ is obtained when $y = \eta$ is the generic point of $Y$.
Suppose there exists $y' \in Y$ such that $d_{y'} < d_\eta$.
After twisting by $\fO_X(-d_{y'})$, we can assume $d_{y'} = 0$.
By (12.8), the function
\begin{equation*}
h^0(y, \fM) = \dim_{k(y)} \cH^0(\P_y, \fO_{\P_y}(d_y))
\end{equation*}
is upper semicontinuous on $Y$ (12.8), and we have the strict inequality
\begin{equation*}
\dim_{k(y')} \cH^0(\P_{y'}, \fO_{\P_{y'}}) < \dim_{k(\eta)} \cH^0(\P_{\eta}, \fO_{\P_{\eta}}(d_\eta))
\end{equation*}
by the explicit calculations of (5.1).
But this implies the set $\{y \in Y \mid h^0(y, \fM) \geq h^0(\eta, \fM)\}$ is a closed set containing $\eta$ and not containing $y'$, a contradiction.
Hence, we have $d_\eta \leq d_{y}$ for all $y \in Y$.

Next, we prove the inequality in the opposite direction.
By Serre duality (5.1) (7.7), we have
\begin{align*}
h^{r-1}(y, \fM) & = \dim_{k(y)} \cH^{r-1}(\P_{y}, \fO_{\P_{y}}(d_y))          \\
& = \dim_{k(y)} \cH^{r-1}(\P_{y}, \fO_{\P_{y}}(-d_y-r)) \\
& = h^0(y, \fO_X(-r) \otimes \fM^{-1}),
\end{align*}
and $h^{r-1}(y, \fM)$ is also an upper semicontinuous on $Y$.
Suppose there exists $y' \in Y$ such that $d_\eta < d_{y'}$.
After twisting by $\fO_X(-d_\eta + r)$, we can assume $d_\eta = r$.
In particular $d_{y'}$ is a positive integer.
Again, we have the strict inequality
\begin{equation*}
\dim_{k(y')} \cH^0(\P_{y'}, \fO_{\P_{y'}}(-d_{y'}-r)) < \dim_{k(\eta)} \cH^0(\P_{\eta}, \fO_{\P_{\eta}}),
\end{equation*}
arriving at the same contradiction as above.
Hence, $d_y$ is independent of $y \in Y$.
\end{proof}
