\begin{enumerate} [label=\textbf{\arabic*.}, leftmargin=0em]

      \item Show that the unit disc in $\C$ is not isomorphic to $X_h$ for any scheme $X$.

      \item Let $z_1, z_2, \dots$ be an infinite sequence of complex numbers with $|z_n| \to \infty$ as $n \to \infty$.
      Let $\gJ \subset \fO_{\C}$ be the sheaf of ideals of holomorphic functions vanishing at all of the $z_n$.
      Show that there is no coherent algebraic sheaf of ideals $\fI \subset \fO_X$, where $X = \A_\C^1$ is the affine line, such that $\gJ = \fI_h$ as an ideal in $\fO_\C$.
      Show on the other hand that there is a coherent sheaf $\fF$ on $X$ such that $\fF_h = \gJ$ as coherent sheaves.

      \item (Serre [12].) On $\C^2 - (0, 0)$, we define an invertible analytic sheaf $\gL$ as follows: $\gL \cong \fO$ when $z \neq 0$, $\gL \cong \fO$ when $w \neq 0$, and when both $z, w \neq 0$, the two copies of $\fO$ are glued by multiplication by $e^{-1/zw}$ in the local ring at the point $(z, w)$.
      Show that there is no invertible algebraic sheaf $\fL$ on $\A^2 - (0, 0)$ with $\fL_h \cong \gL$.

      \item Show directly that if $X$ is a scheme which is reduced and proper over $\C$, then $\cH^0(X, \fO_X) \cong \cH^0(X_h, \fO_{X_h})$.
      Conversely, show that if $X$ is not proper over $\C$, then there is a coherent sheaf $\fF$ on $X$ with $\cH^0(X, \fF) \neq \cH^0(X_h, \fF_h)$.

      \item If $X, X'$ are nonsingular affine algebraic curves, with $X_h \cong X_h'$, show that $X \cong X'$.

      \item Show that if $X$ and $Y$ are projective schemes over $\C$, and $\gf : X_h \to Y_h$ is a morphism of analytic spaces, then there exists a (unique) morphism $f : X \to Y$ with $f_h = \gf$.

\end{enumerate}
