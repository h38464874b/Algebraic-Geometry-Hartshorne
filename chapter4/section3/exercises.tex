\begin{enumerate} [label=\textbf{\arabic*.}, leftmargin=0em]

  \item If $X$ is a curve of genus 2, show that a divisor $D$ is very ample $\iff$ $\deg{D} \geq 5$.
        This strengthens (3.3.4).

        \begin{proof}
          Let $D$ be a very ample divisor on $X$.
          By the degree-genus formula, a nonsingular plane curve cannot have genus equal to 2, which implies $\dim |D| \geq 3$.
          Applying the Riemann-Roch theorem to $D$ gives
          \begin{equation*}
            \deg{D} = \dim |D| - \dim |K - D| + 1 \geq 4 - \dim|K - D|.
          \end{equation*}
          By (Ex. 1.5), $D$ has degree at least 3 (Ex. 1.5), so $K - D$ has negative degree.
          Hence, $|K - D|$ has dimension -1.
        \end{proof}

  \item[\textbf{3.}] If $X$ is a curve of genus $\geq 2$ which is a complete intersection (II, Ex. 8.4) in some $\P^n$, show that the canonical divisor $K$ is very ample.
        Conclude that a curve of genus 2 can never be a complete intersection in any $\P^n$.

        \begin{proof}
          Let $d$ be the degree of $X$ in $\P^n$, and assume $X$ is not contained in any hyperplane.
          Then $X$ is a complete intersection of $n - 1$ hypersurfaces of degree $d_i$, and by (II, Ex. 8.4), we have $\fO_X(K) \cong \fO_X(e)$, where $e = \sum d_i - n - 1$, so it is enough to show $e \geq 1$.
          We have two ways of computing $\deg{K}$, which are $\deg{K} = ed$ and $\deg{K} = 2g - 2 \geq 2$.
          Hence, $e \geq 1$.
          The linear system given by the canonical sheaf of a curve of genus 2 is the unique $g^1_2$ (Ex. 1.7), which is never ample.
        \end{proof}

  \item[\textbf{6.}] \textit{Curves of Degree 4.}
        \begin{itemize}
          \item[(a)] If $X$ is a curve of degree 4 in some $\P^n$, show that either
                \begin{itemize}
                  \item[(1)] $g = 0$, in which case $X$ is either the rational normal quartic in $\P^4$ (Ex. 3.4) or the rational quartic curve in $\P^3$ (II, 7.8.6), or
                  \item[(2)] $X \subseteq \P^2$, in which case $g = 3$, or
                  \item[(3)] $X \subseteq \P^3$ and $g = 1$.
                \end{itemize}
          \item[(b)] In the case $g = 1$, show that $X$ is a complete intersection of two irreducible quadric surfaces in $\P^3$ (I, Ex. 5.1).
        \end{itemize}

        \begin{proof} $ $ \vspace{0pt}
          \begin{itemize} [leftmargin=0cm]
            \item[(a)] \fbox{$n = 2$} The degree-genus formula gives
                  \begin{equation*}
                    g = \frac{(d - 1)(d - 2)}{2} = 3.
                  \end{equation*}

                  \fbox{$n = 3$} If $X$ is not contained in any hyperplane, then $g < 3$ (III, Ex. 9.11) (Ex. 3.5b).
                  Since $g \neq 2$ by (Ex. 3.1), $X$ is either the rational quartic curve in $\P^3$, or an elliptic curve.

                  \fbox{$n = 4$} $X$ is rational by (Ex. 1.5).

            \item[(b)] Let $\fI$ be the ideal sheaf of $X$ in $\P^3$, and consider the exact sequence
                  \[ \begin{tikzcd}
                      0 \arrow[r] & \fI \arrow[r] & \fO_{\P^3} \arrow[r] & \fO_X \arrow[r] & 0.
                    \end{tikzcd} \]
                  Twisting by $\fO_{\P_3}(2)$ and taking cohomology gives
                  \[ \begin{tikzcd}
                      0 \arrow[r] & \cH^0(\P^3, \fI(2)) \arrow[r] & \cH^0(\P^3, \fO_{\P^3}(2)) \arrow[r] & \cH^0(X, \fO_X(2)) \arrow[r] & \cdots.
                    \end{tikzcd} \]
                  Now, $X$ is an elliptic curve, and $\fO_X(2)$ corresponds to a divisor $D$ of degree 8.
                  The Riemann Roch theorem gives $\dim \cH^0(X, \fO_X(2)) = 8$, and the vector space in the middle has dimension 10, so we conclude that $h^0(\P^3, \fI(2)) \geq 2$.
                  An element of $\cH^0(\P^3, \fI(2))$ is a form of degree 2, whose zero-set is a quadric surface containing $X$.
                  Any such quadric must be irreducible and reduced because $X$ is not contained in any hyperplane.
                  Hence, $X$ is contained in two distinct irreducible quadric surfaces and is a complete intersection because $X$ is a curve of degree 4.
          \end{itemize}
        \end{proof}

  \item[\textbf{9.}] Prove the following lemma of Bertini: if $X$ is a curve of degree $d$ in $\P^3$, not contained in any plane, then for almost all planes $H \subseteq \P^3$ (meaning a Zariski open subset of the dual projective space $(\P^3)^*$), the intersection $X \cap H$ consists of exactly $d$ distinct points, no three of which are collinear.

        \begin{proof}
          Bertini's theorem says a general plane in $\P^3$ intersects $X$ in $d$ distinct points.
          To show that no three of these points are collinear generically, it is enough to show a general plane in $\P^3$ does not contain a multisecant of $X$.
          The proof of (3.10) shows that the set of multisecants of $X$ has dimension $\leq 1$ in $(\P^3)^*$, and a general plane in $\P^3$ corresponds to a line in $(\P^3)^*$.
          Hence, for almost almost all planes $H \subseteq \P^3$, the intersection $X \cap H$ consists of exactly $d$ distinct points, no three of which are collinear.
        \end{proof}

  \item[\textbf{12.}] For each value of $d = 2, 3, 4, 5$ and $r$ satisfying $0 \leq r \leq \frac{1}{2}(d - 1)(d - 2)$, show that there exists an irreducible plane curve of degree $d$ with $r$ nodes and no other singularities.

        \begin{proof}
          There is nothing to prove for $d = 2$, so fix any $d \geq 3$, and denote $g = \frac{1}{2}(d - 1)(d - 2)$.
          \begin{equation*}
            d = 2, 3, 4, 5, \quad g = 0, 1, 3, 6
          \end{equation*}

          Both extremes $r = 0, g$ are consructed in (3.11.1).

          If $X$ is any elliptic curve and $P$ is any point in $X$, then the linear system $|dP|$ gives an embedding of $X$ in $\P^{d - 1}$ as a curve of degree $d$.
          Projecting the curve into $\P^2$ by (3.5) and (3.10) gives a plane curve of degree $d$ and $g - 1$ nodes.
          This gives $d = 4$ and $r = 2$.

          By (Ex 3.6), the complete intersection of two quadric surfaces in $\P^3$ is an elliptic curve of degree 4.
          Projecting from a point not in the curve by (3.10) gives a plane curve of degree $d = 4$ and $r = 2$ nodes

          Similarly, any divisor of degree $\geq 5$ on a genus 2 curve is very ample by (Ex. 1.5), so there exists plane curves of degree $d \geq 5$ with $g - 2$ nodes.
          This gives $d = 5, r = 4$.

          If $Q \subset \P^3$ is the nonsingular quadric suface, then there exists divisors of type $(a, b)$ for any $a, b > 0$ that are linearly equivalent to an irreducible nonsingular curve of degree $a + b$ and genus $(a - 1)(b - 1)$ as a curve in $\P^3$.

          A divisor of type $(2, 3)$ gives a nonsingular curve of degree $2 + 3 = 5$ and genus $(2 - 1)(3 - 1) = 2$ in $\P^3$.
          Projecting into $\P^2$ by (3.10) gives a plane curve of degree $d = 5$ and $r = 4$ nodes.
          Projecting this curve from one of its own points that does not lie in a plane spanned by coplanar tangent lines gives a plane curve of degree $d = 4$ and $r = 1$ node.

          Now, consider a divisor of type $(3, 3)$.
          It is a nonsingular curve of degree $3 + 3 = 6$ and genus $(3 - 1)(3 - 1) = 4$ in $\P^3$.
          Projecting this curve from one of its points that does not lie in a plane spanned by coplanar tangent lines gives a plane curve of degree $d = 5$ and $r = 2$ nodes.

          Finally, consider the divisor of type $(4, 2)$.
          It has degree $2 + 4 = 6$ and genus $(4 - 1)(2 - 1) = 3$ as a curve in $\P^3$.
          Projecting this curve from one of its points that does not lie in a plane spanned by coplanar tangent lines gives a plane curve of degree $d = 5$ and $r = 3$ nodes.

          More generally, any divisor of degree $d \geq 2g + 1$ on any curve of genus $g$ is very ample by (3.2), so there exists plane curves of degree $d$ and $g - d$ nodes.
          This gives $d = 5$ and $r = 1$.

        \end{proof}

\end{enumerate}
