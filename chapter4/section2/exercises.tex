  \item Use (2.5.3) to show that $\P^n$ is simply connected.

        \begin{proof}
          Proceed by induction on $n$.
          The base case $n = 1$ is (2.5.3).
          Let $f : X \to \P^n$ be an étale covering.
          We may assume $X$ is connected.
          Then $X$ is smooth over $k$ since $f$ is étale (III, 10.1), and $X$ is projective over $k$ since $f$ is finite, so $X$ is a smooth irreducible projective variety of dimension $n$.
          It is enough to show the degree of $f$ is 1.
          By (III, Ex. 11.3), $f^{-1}(H) = X \times_{\P^n} H$ is smooth and irreducible for almost every hyperplane $H \subset \P^n$.
          Also, the induced morphism $f' : f^{-1}(H) \to H$ is an étale covering of $H \cong \P^{n - 1}$, since étale morphisms are stable under base change.
          By the inductive hypothesis, $f'$ is an isomorphism.
          Hence, $f$ is an isomorphism.
        \end{proof}

  \item \textit{Classification of Curves of Genus 2.} Fix an algebraically closed field $k$ of characteristic $\neq 2$.
        \begin{enumerate} [label=(\alph*)]
          \item If $X$ is a curve of genus 2 over $k$, the canonical linear system $|K|$ determines a finite morphism $f : X \to \P^1$ of degree 2 (Ex. 1.7). Show that it is ramified at exactly 6 points, with ramification index 2 at each one.
                Note that $f$ is uniquely determined, up to an automorphism of $\P^1$, so $X$ determines an (unordered) set of 6 points of $\P^1$, up to an automorphism of $\P^1$.

          \item Conversely, given six distinct elements $\alpha_1, \dots, \alpha_6 \in k$, let $K$ be the extension of $k(x)$ determined by the equation $z^2 = (x - \alpha_1) \cdots (x-\alpha_6)$.
                Let $f : X \to \P^1$ be the corresponding morphism of curves.
                Show that $g(X) = 2$, the map $f$ is the same as the one determined by the canonical linear system, and $f$ is ramified over the six points $x = \alpha_i$ of $\P^1$, and nowhere else.

          \item Using (I, Ex. 6.6), show that if $P_1, P_2, P_3$ are three distinct points of $\P^1$, then there exists a unique $\varphi \in \Aut{\P^1}$ such that $\varphi(P_1) = 0$, $\varphi(P_2) = 1$, $\varphi(P_3) = \infty$.
                Thus in (a), if we order the six points of $\P^1$, and then normalize by sending the first three to $0, 1, \infty$, respectively, we may assume that $X$ is ramified over $0, 1, \infty, \beta_1, \beta_2, \beta_3$, where $\beta_1, \beta_2, \beta_3$ are three distinct elements of $k$, $\neq 0, 1$.

          \item Let $\Sigma_6$ be the symmetric group on 6 letters.
                Define an action of $\Sigma_6$ on sets of three distinct elements $\beta_1, \beta_2, \beta_3$ of $k$, $\neq 0, 1$, as follows: reorder the set $0, 1, \infty, \beta_1, \beta_2, \beta_3$ according to a given element $\sigma \in \Sigma_6$, then renormalize as in (c) so that the three become $0, 1, \infty$ again.
                Then the last three are the new $\beta_1', \beta_2', \beta_3'$.

          \item Summing up, conclude that there is a one-to-one correspondence between the set of isomorphism classes of curves of genus 2 over $k$, and triples of distinct elements $\beta_1, \beta_2, \beta_3$ of $k$, $\neq 0, 1$, modulo the action of $\Sigma_6$ described in (d).
                In particular, there are many non-isomorphic curves of genus 2.
                We say that curves of genus 2 depend on three parameters, since they correspond to the points of an open subset of $\A_k^3$ modulo a finite group.
        \end{enumerate}

        \begin{proof} $ $ \vspace{0pt}
          \begin{enumerate} [leftmargin=0cm]
            \item By Hurtwitz's theorem, the degree of the ramification divisor is computed to be
                  \begin{equation*}
                    \deg{R} = \deg{K_X} - 2 \deg{K_{\P^1}} = 6.
                  \end{equation*}
                  The ramification index at a point cannot exceed the degree of the morphism, so $f$ must be ramified at exactly 6 distinct points each having ramification index 2.

            \item Let $B = k[x, z]/(z^2 - g(x))$ be an affine coordinate ring of $X$ with $g(x) = (x - \alpha_1) \cdots (x - \alpha_6)$, and let $f$ be the projection morphism onto the $x$-axis.
                  By (II, 8.3A), we have
                  \begin{equation*}
                    \Omega_{B / k} \cong (B \dd x \oplus B \dd z) / (g'(x) \dd x + 2z \dd z).
                  \end{equation*}
                  Let $P = (x', z')$ be a closed point in $X$.
                  If $z' \neq 0$, then $z$ is invertible in $\fO_{P, X}$, so that
                  \begin{equation*}
                    \dd z = - \frac{g'(x)}{2z} \dd x,
                  \end{equation*}
                  which means $\dd x$ generates $(\Omega_{B / k})_P$.
                  If $z' = 0$, then $x' = \alpha_i$ for some $i$, and $g'(x)$ is invertible in $\fO_{P, X}$, so that
                  \begin{equation*}
                    \dd x = - \frac{2z}{g'(x)} \dd z,
                  \end{equation*}
                  which means $\dd z$ generates $(\Omega_{B / k})_P$.
                  Since $\dd x $ is a local parameter of $\P^1$, we see that $P$ is ramified if and only if $z' = 0$ with ramification index $e_P = v_P( \dd x / \dd z) + 1 = 2$.
                  There are six ramification points corresponding to the six roots of $g(x)$.
                  Hence, $g(X) = 2$ by the Hurwitz formula.
          \end{enumerate}
        \end{proof}

  \item[\textbf{3.}] \textit{Plane Curves.} Let $X$ be a curve of degree $d$ in $\P^2$.
        For each $P \in X$, let $T_P(X)$ be the tangent line to $X$ at $P$ (I, Ex.7.3).
        Considering $T_P(X)$ as a point of the dual projective plane $(\P^2)^*$, the map $P \to T_P(X)$ gives a morphism of $X$ to its \textit{dual curve} $X^*$ in $(\P^2)^*$ (I, Ex. 7.3).
        Note that even though $X$ is nonsingular, $X^*$ in general will have singularities.
        We assume $\car{k} = 0$ below.

        \begin{itemize}
          \item[(a)] Fix a line $L \subseteq \P^2$ which is not tangent to $X$.
                Define a morphism $\varphi : X \to L$ by $\varphi(P) = L \cap T_P(X)$, for each point $P \in X$.
                Show that $\varphi$ is ramified at $P$ if and only if either (1) $P \in L$, or (2) $P$ is an \textit{inflection point} of $X$, which means that the intersection multiplicity (I, Ex. 5.4) of $T_P(X)$ with $X$ at $P$ is $\geq 3$.
                Conclude that $X$ has only finitely many inflection points.

          \item[(b)] A line of $\P^2$ is a \textit{multiple tangent} of $X$ if it is tangent to $X$ at more than one point.
                It is a \textit{bitangent} if it is tangent to $X$ at exactly two points.
                If $L$ is a multiple tangent of $X$, tangent to $X$ at the points $P_1, \dots, P_r$, and if none of the $P_i$ is an inflection point, show that the corresponding point of the dual curve $X^*$ is an \textit{ordinary $r$-fold point}, which means a point of multiplicity $r$ with distinct tangent directions (I, Ex. 5.3).
                Conclude that $X$ has only finitely many multiple tangents.

          \item[(c)] Let $O \in \P^2$ be a point which is not on $X$, nor on any inflectional or multiple tangent of $X$.
                Let $L$ be a line not containing $O$.
                Let $\psi : X \to L$ be the morphism defined by projection from $O$.
                Show that $\psi$ is ramified at $P \in X$ if and only if the line $OP$ is tangent to $X$ at $P$, and that the ramification index is 2.
                Use Hurwitz's theorem and (I, Ex. 7.2) to conclude that there are exactly $d(d - 1)$ tangents of $X$ passing through $O$.
                Hence, the degree of the dual curve (sometimes called the \textit{class} of $X$) is $d(d - 1)$.

          \item[(d)] Show that for all but a finite number of points of $X$, a point $O$ of $X$ lies on exactly $(d + 1)(d - 2)$ tangents of $X$, not counting the tangent at $O$.

          \item[(e)] Show that the degree of the morphism $\varphi$ of (a) is $d(d - 1)$.
                Conclude that if $d \geq 2$, then $X$ has $3d(d - 2)$ inflection points, properly counted.
                (If $T_P(X)$ has intersection multiplicity $r$ with $X$ at $P$, then $P$ should be counted $r - 2$ times as an inflection point.
                If $r = 3$, we called it an \textit{ordinary inflection point}.)
                Show that an ordinary inflection point of $X$ corresponds to an ordinary cusp of the dual curve $X^*$.

          \item[(f)] Now let $X$ be a plane curve of degree $d \geq 2$, and assume that the dual curve $X^*$ has only nodes and ordinary cusps as singularities (which should be true for sufficiently general $X$.)
                Then show that $X$ has exactly $\frac{1}{2}d(d - 2)(d - 3)(d + 3)$ bitangents.

          \item[(g)] For example, a plane cubic curve has exactly 9 inflection points, all ordinary.
                The line joining any two of them intersects the curve in a third one.

          \item[(h)] A plane quartic curve has exactly 28 bitangents.
                (This holds even if the curve has a tangent with four-fold contact, in which case the dual curve $X^*$ has a tacnode.)
        \end{itemize}

        \filbreak

        \begin{proof} $ $ \vspace{0pt}
          \begin{itemize} [leftmargin=0cm]
            \item[(a)] Find local affine coordinates $x, y$ of $\P^2$ such that $$X : y - f(x) = 0, \quad L : x = 0.$$
                  Setting $x$ and $y$ to be local parameters for $X$ at $P$ and $L$ at $\varphi(P)$, respectively, we can write $\varphi$ in the form
                  \begin{equation*}
                    \varphi(x) = f(x) - xf'(x).
                  \end{equation*}
                  It follows that $$\varphi^* \dd y = \frac{\dd y}{\dd x} \dd x, \quad \frac{\dd y}{\dd x} = -xf''.$$
                  By (2.2), $\varphi$ is ramified at $P$ if and only if $v_P(\dd t / \dd u) > 0$, where we have
                  \begin{equation*}
                    v_P\bigg(\dv{y}{x}\bigg) = v_P(x) + v_P(f'').
                  \end{equation*}
                  Notice that $v_P(x) > 0$ if and only if $P \in L$.
                  Also, it is not hard to see
                  \begin{equation*}
                    \mu_P(X, T_P(X)) = v_P(f'') + 2.
                  \end{equation*}
                  Thus, $v_P(\dd t / \dd u) > 0$ if and only if $v_P(x) > 0$ or $v_P(f'') > 0$ if and only if (10 )$P \in L$ or (2) $P$ is an inflection point of $X$.
                  Since $\varphi : X \to L$ ramifies at finitely many points, $X$ has finitely many inflection points.

            \item[(b)] Find local affine coordinates $x, y$ of $\P^2$ such that
                  \begin{equation*}
                    X : y - f(x) = 0, \quad L : y = 0.
                  \end{equation*}
                  This means $x$ is a local parameter of $X$, so we can write $\gamma : X$ $\ : X \to X^* \subset (\P^2)^*$ in the form
                  \begin{equation*}
                    \gamma(x) = (-f'(x), xf'(x) - f(x)).
                  \end{equation*}
                  The tangent directions of $Q = \gamma(P_i)$ are precisely the images of the tangent vectors at $P_i$ under the differential of $\gamma$.
                  In particular, identifying the tangent vector of $X$ at $P_i$ to the row vector $(1, 0) \in T_{P_i}(X)$ and applying $\dd \gamma$ gives
                  \begin{equation*}
                    v_i = (-f''(x_i), x_if''(x_i)) \in T_Q(X^*).
                  \end{equation*}
                  We have shown in (a) that $f''(x_i) = 0$ if and only if $P_i$ is an inflection point, which implies $v_i$ and $v_j$ define the same direction if and only if $P_i = P_j$.
                  Hence, $Q$ is an ordinary $r$-fold point with distinct tangent directions.
                  Each multiple tangent of $X$ corresponds to singular point of $X^*$, but a curve can have only finitely many singular points.
                  Hence, $X$ has finitely many multiple tangents.

            \item[(c)] Find local affine coordinates $x, y$ of $\P^2$ such that
                  \begin{equation*}
                    X : y - f(x) = 0, \quad L : x = 0, \quad O = (1 : 0 : 0).
                  \end{equation*}
                  We can set $u = x$ and $t = y$ to be local parameters for $X$ at $P$ and $L$ at $Q = \psi(P)$, respectively, and $\psi$ is given by the projection morphism onto the $x$-axis.
                  It follows that
                  \begin{equation*}
                    \psi^* \dd y = \frac{\dd y}{\dd x} \dd x, \quad \frac{\dd y}{\dd x} = f'.
                  \end{equation*}
                  Hence, $\psi$ is ramified at $P \in X$ if and only if $f'(P) = 0$ if and only if the tangent of $X$ at $P$ passes through $O$.
                  Suppose $\psi$ is ramified at $P$.
                  Since $O$ is not on any inflectional or multiple tangents of $X$, $v_P(f'') = 0$.
                  Hence, $v_P(f') = v_P(f'') + 1 = 1$.

                  The genus of $X$ is related to $d$ by the degree genus formula.
                  Since $\psi$ has degree $d$ as a morphism of curves, Hurwitz's theorem can be used to compute the degree of the ramification divisor
                  \begin{equation*}
                    \deg{R} = \deg{K_X} - d \deg{K_{\P^1}} = d(d - 1).
                  \end{equation*}
                  The ramification index of each ramification point is 2, so the degree of $R$ equals to the number of tangents passing through $O$.

            \item[(d)] This is another application of Hurwitz's theorem.
                  The projection morphism $\psi : X - O \to L$ has degree $d - 1$, so the degree of the ramification divisor of $\psi$ is
                  \begin{equation*}
                    \deg{R} = \deg K_X - (d - 1) \deg{K_{\P^1}} = (d + 1)(d - 2).
                  \end{equation*}
                  A general point of $X$ does not lie on any multiple tangents of $X$.
                  Hence, the degree of $R$ equals to the number of tangents passing through $O$.

            \item[(e)] Define $\pi : (\P^2)^* - L \to L$ by $L' \mapsto L \cap L'$.
                  Since $\varphi$ factors into $\varphi = \pi \circ \gamma$, the degree of $\varphi$ equals to the degree of the image of $\gamma$, which is $d(d - 1)$.
                  The degree of the ramification divisor of $\varphi$ is
                  \begin{equation*}
                    \deg{R} = \deg K_X - d(d - 1) \deg{K_{\P^1}} = 3d^2 - 5d.
                  \end{equation*}
                  The degree of the ramification index also counts the number of points $X \cap L$.
                  In particular, $X \cap L$ is generically $d$ distinct points, and the ramification index of each $P \in X \cap L$ is 2, so we need to subtract $d$ from $\deg{R}$ to get the total number of inflection points of $X$, which equals to $3d(d - 2)$.
                  If $r = 3$, an ordinary inflection point of $X$ corresponds to a twisted cubic in the dual curve locally, which has an ordinary cusp.

            \item[(f)] Let $\beta_1$ and $\beta_2$ be the number of bitangents and inflection points of $X$, respectively.
                  We have two ways of computing the arithmetic genus of $X^*$: (1) in terms of $p_a(X)$, and (2) $X^* \subset (\P^2)^*$ as a plane curve of degree $d(d - 1)$.

                  (1) The morphism $\gamma : X \to X^*$ explicitly defined in (b) is generically one-to-one by, so $\gamma$ is a birational onto its image.
                  The results of (Ex. 1.8) say
                  \begin{equation*}
                    p_a(X^*) = p_a(X) + \delta, \quad \delta = \sum_{P \in X^*} \delta_P,
                  \end{equation*}
                  where $\delta_P = 1$ if $P \in X^*$ is a node or ordinary cusp, zero otherwise.
                  In particular, $\delta = \beta_1 + \beta_2$ by (c) and (e).

                  (2) On the otherhand, $X^*$ is a plane curve of degree $d(d - 1)$ in $(\P^2)^*$.
                  The degree-genus formula for plane curves (III, Ex. 4.7) gives
                  \begin{equation*}
                    p_a(X^*) = \frac{1}{2}(d - 2)(d + 1) (d^2 - d - 1)
                  \end{equation*}

                  A smooth plane curve of degree $d$ has $d(d - 2)$ inflection points.
                  Hence, the number of bitangets of $X$ is
                  \begin{equation*}
                    \beta_1 = p_a(X^*) - p_a(X) - \beta_2 = \frac{1}{2}d(d - 2)(d - 3)(d + 3).
                  \end{equation*}

            \item[(g)] Let $E \subset \P^2$ be a plane cubic curve, let $P, Q \in E$ be two inflection points, and let $L$ the line through $P$ and $Q$.
                  If $P = Q$, then $L$ is the tangent line through $P$.
                  In this case, $P$ is an inflection point, so it is counted with a multiplicity of 3, i.e., $L$ meets no other point of $X$.
                  Otherwise, $L$ cannot be a tangent line for the same reason, and $L$ meets $X$ at a distinct point $R \neq P, Q$.
                  The morphism $\varphi : E \to L$ defined in (a) is ramified at $R$.
                  By (e), $\varphi$ has degree $=3(3 - 1) = 6$.
                  Since $P$ and $Q$ are are ordinary inflection points, we have the ramification index of $\varphi$ at $P$ and $Q$ is 2.
                  Hence, $R$ also has ramification index of 2, which is an inflection point.
          \end{itemize}
        \end{proof}

  \item[\textbf{5.}] \textit{Automorphisms of a Curve of Genus $\geq 2$.} Prove the theorem of Hurtwitz [1] that a curve $X$ of genus $g \geq 2$ over a field of characteristic 0 has at most $84(g - 1)$ automorphisms.
        We will see later (Ex. 5.2) or (V, Ex. 1.11) that the group $G = \Aut{X}$ is finite.
        So let $G$ have order $n$.
        Then $G$ acts on the function field $K(X)$.
        Let $L$ be the fixed field.
        Then the field extension $L \subseteq K(X)$ corresponds to a finite morphism of curves $f : X \to Y$ degree $n$.
        \begin{itemize}
          \item[(a)] If $P \in X$ is a ramification point, and $e_p = r$, show that $f^{-1}f(P)$ consists of exactly $n/r$ points, each having ramification index $r$.
                Let $P_1, \dots, P_s$ be a maximal set of ramification points of $X$ lying over distinct points of $Y$, and let $e_{P_i} = r_i$.
                Then show that Hurtwitz's theorem implies that
                \begin{equation*}
                  (2g - 2)/n = 2g(Y) - 2 + \sum_{i = 1}^s (1 - 1/r_i).
                \end{equation*}
          \item[(b)] Since $g \geq 2$, the left hand side of the equation is $> 0$.
                Show that if $g(Y) \geq 0$, $s \geq 0$, $r_i \geq 2$, $i = 1, \dots, s$ are integers such that
                \begin{equation*}
                  2g(Y) - 2 + \sum_{i = 1}^s (1 - 1 / r_i) > 0,
                \end{equation*}
                then the minimum value of this expression is $1 / 42$.
                Conclude that $n \leq 84(g - 1)$.
        \end{itemize}

        \begin{proof} $ $ \vspace{0pt}
          \begin{enumerate} [leftmargin=0cm]
            \item[(a)] Each $\sigma \in G$ induces an automorphism of $X$ such that $f \circ \sigma = f$, so there is a group action of $G$ on each fibre $f^{-1}f(P)$.
                  The action is transitive (A.M., Ex. 5.13), and $e_P = e_{\sigma(P)}$ because $\sigma$ induces an isomorphism of local rings $\fO_{\sigma(P_i), X} \to \fO_{P_i, X}$, hence sends uniformizer to uniformizer.
                  In particular, if $d_i$ is the size of the orbit of $P_i$, then $n = d_i r_i$.
                  Hence, we have
                  \begin{equation*}
                    \sum_{P \in X} (e_P - 1) = \sum_{i = 1}^s \sum_{P \in f^{-1}f(P_i)} (e_P - 1) = \sum_{i = 1}^s d_i(r_i - 1)                      = n \sum_{i = 1}^s (1 - 1/r_i).
                  \end{equation*}

            \item[(b)] The following give the minimum value
                  \begin{equation*}
                    g(Y) = 0, \quad s = 3, \quad r_1 = \frac{1}{2}, \quad r_2 = \frac{1}{3}, \quad r_3 = \frac{1}{7}.
                  \end{equation*}
                  We conclude that
                  \begin{equation*}
                    n \leq 42(2g - 2) = 84(g - 1).
                  \end{equation*}
          \end{enumerate}
        \end{proof}

  \item[\textbf{6.}] \textit{$f_*$ for Divisors.} Let $f : X \to Y$ be a finite morphism of curves of degree $n$.
        We define a homomorphism $f_* : \Div{X} \to \Div{Y}$ by $f_*(\sum n_i P_i) = \sum n_i f(P_i)$ for any divisor $D = \sum n_i P_i$ on $X$.
        \begin{itemize}
          \item[(a)] For any locally free sheaf $\fE$ on $Y$, of rank $r$, we define $\det \fE = \bigwedge^r \fE \in \pic{Y}$ (II, Ex. 6.11).
                In particular, for any invertible sheaf $\fM$ on $X$, $f_*\fM$ is locally free of rank $n$ on $Y$, so we can consider $\det f_*\fM \in \pic{Y}$.
                Show that for any divisor $D$ on $X$,
                \begin{equation*}
                  \det(f_*\fO_X(D)) \cong (\det f_*\fO_X) \otimes \fO_Y(f_*D).
                \end{equation*}
                Note in particular that $\det(f_*\fO_X(D)) \neq \fO_Y(f_*D)$ in general!.
          \item[(b)] Conclude that $f_*D$ depends only on the linear equivalence class of $D$, so there is an induced homomorphism $f_* : \pic{X} \to \pic{Y}$.
                Show that $f_* \circ f^* : \pic{Y} \to \pic{Y}$ is just multiplication by $n$.
          \item[(c)] Use duality for a finite flat morphisms (III, Ex. 6.10) and (III, Ex. 7.2) to show that
                \begin{equation*}
                  \det f_*\Omega_X \cong (\det f_*\fO_X )^{-1} \otimes \Omega_Y^{\otimes n}.
                \end{equation*}
          \item[(d)] Now assume that $f$ is separable, so we have the ramification divisor $R$.
                We define the \textit{branch divisor} $B$ to be the divisor $f_*R$ on $Y$.
                Show that
                \begin{equation*}
                  (\det f_*\fO_X)^2 \cong \fO_Y(-B).
                \end{equation*}
        \end{itemize}

        \begin{proof} $ $ \vspace{0pt}
          \begin{itemize} [leftmargin=0cm]
            \item[(a)] Let $P$ be any point in $X$, and let $f_*P = f(P)$ be the image of $P$. 
                  The skyskraper sheaf $k(f_*P)$ has a natural free resolution
                  \[ \begin{tikzcd}
                      0 \arrow[r] & \fO_Y(-f_*P) \arrow[r] & \fO_Y \arrow[r] & k(f_*P) \arrow[r] & 0.
                    \end{tikzcd} \]
                  In particular, $\det k(f_*P) \cong \fO_Y(f_*P)$, and any finite free resolution of $k(f_*P)$ induces isomorphic determinant line bundles (II, Ex. 6.11).
                  Also, the sheaves $k(f_*P) \to f_*k(P)$ are isomorphic, so any free resolution of $f_*k(P)$ is also a free resolution of $k(f_*P)$.

                  Now, let $D$ be any non-zero divisor, and let $P$ be any point.
                  We will show that the formula is true for $D$ if and only if it is true for $D + P$ (1.3).
                  Consider the exact sequence
                  \[ \begin{tikzcd}
                      0 \arrow[r] & \fO_X(-P) \arrow[r] & \fO_X \arrow[r] & k(P) \arrow[r] & 0.
                    \end{tikzcd} \]
                  Tensoring with $\fO_X(D + P)$ and applying $f_*$ gives
                  \[ \begin{tikzcd}
                      0 \arrow[r] & f_*\fO_X(D) \arrow[r] & f_*\fO_X(D + P) \arrow[r] & f_*k(P) \arrow[r] & 0.
                    \end{tikzcd} \]
                  By the remarks above, we have an isomorphism
                  \begin{equation*}
                    \fO_Y(f_*P) \cong \det(f_*\fO_X(D+P)) \otimes (\det(f_*\fO_X(D)))^{-1},
                  \end{equation*}
                  which is what we wanted to show.

            \item[(b)] By (a), since $f_*\fO_X(D)$ depends only on the linear equivalence class of $D$, so does $\fO_Y(f_*D)$.
                  For the second part, it suffices to show the statement holds for any point $Q \in Y$.
                  Indeed, it follows by definition $f_* \circ f^*$ is multiplication by some integer.
                  On the other hand, $f^*$ multiplies the degree by $n$ (II, 6.9), and $f_*$ preserves degrees.
                  Hence, $f_* \circ f^*$ must be multiplication by $n$.

            \item[(c)] Let $\fF$ be a coherent sheaf on $Y$.
                  By definition, we have
                  \begin{equation*}
                    f_*f^!\fF \cong f_*\fO_X^{-1} \otimes \fF, \quad f_*\fO_X^{-1} = \rHom_Y(f_*\fO_X, \fO_Y).
                  \end{equation*}
                  By (III, Ex. 7.3), $f^! \Omega_Y \cong \Omega_X$ is a dualizing sheaf on $X$.
                  Applying above to $\fF = \Omega_Y$ gives
                  \begin{align*}
                    \det f_*\Omega_X & \cong \det f_*f^!\Omega_Y                                \\
                                     & \cong \det(f_*\fO_X^{-1} \otimes \Omega_Y)               \\
                                     & \cong (\det f_*\fO_X)^{-1} \otimes \Omega_Y^{\otimes n}.
                  \end{align*}

            \item[(d)] Let $K_X$ and $K_Y$ be canonical divisors of $X$ and $Y$, respectively.
                  They satisfy the following relation (2.3)
                  \begin{equation*}
                    f^*K_Y - K_X \sim -R.
                  \end{equation*}
                  Applying $f_*$ gives
                  \begin{equation*}
                    nK_Y - f_*K_X \sim - B.
                  \end{equation*}
                  On the otherhand, (a) applied to $D = K_X$ gives
                  \begin{equation*}
                    \fO_Y(-f_*K_X) \cong (\det f_*\fO_X) \otimes (\det f_*\Omega_X)^{-1}.
                  \end{equation*}
                  and by (c), we have the isomorphism
                  \begin{align*}
                    \fO_Y(nK_Y) \cong (\det f_*\fO_X) \otimes (\det f_*\Omega_X),
                  \end{align*}
                  Combining these two isomorphisms gives
                  \begin{equation*}
                    (\det f_*\fO_X)^2 \cong \fO_Y(nK_Y - f_*K_X) \cong \fO_Y(-B).
                  \end{equation*}
          \end{itemize}
        \end{proof}

  \item[\textbf{7.}] \textit{Étale Covers of Degree 2.} Let $Y$ be a curve over a field $k$ of characteristic $\neq 2$.
        We show there is a one-to-one correspondence between finite étale morphisms $f : X \to Y$ of degree 2, and 2-torsion elements of $\pic{Y}$, i.e., invertible sheaves $\fL$ on $Y$ with $\fL^2 \cong \fO_Y$.
        \begin{itemize}
          \item[(a)] Given an étale morphism $f : X \to Y$ of degree 2, there is a natural map $\fO_Y \to f_*\fO_X$.
                Let $\fL$ be the cokernel.
                Then $\fL$ is an invertible sheaf on $Y$, $\fL \cong \det f_*\fO_X$, and so $\fL^2 \cong \fO_X$ by (Ex. 2.6).
                Thus an étale cover of degree 2 determines a 2-torsion element in $\pic{Y}$.
          \item[(b)] Conversely, given a 2-torsion element $\fL$ in $\pic{Y}$, define an $\fO_Y$-algebra structure on $\fO_Y \oplus \fL$ by $\langle a, b \rangle \cdot \langle a', b' \rangle = \langle aa' + \varphi(b \otimes b'), ab' + a'b \rangle$, where $\varphi$ is an isomorphism of $\fL \otimes \fL \to \fO_Y$.
                Then take $X = \rSpec(\fO_Y \oplus \fL)$.
                Show that $X$ is an étale cover of $Y$.
          \item[(c)] Show that these two processes are inverse to each other.
        \end{itemize}

        \begin{proof} $ $ \vspace{0pt}
          \begin{itemize} [leftmargin=0cm]
            \item[(a)] We check that $\fL$ is an invertible sheaf on $Y$.
                  At the level of stalks, the natural map $\fO_{Q, Y} \to (f_*\fO_X)_Q \cong \fO_{Q, Y}^{\oplus 2}$ is the diagonal map, so the cokernel is free of rank one.
                  Hence, $\fL$ is locally free of rank one.

            \item[(b)] By construction, $f : X \to Y$ is a finite morphism, and $f_*\fO_X \cong \fO_Y \oplus \fL$ is locally free of rank 2, so $f$ is a flat morphism.
                  Also, $\varphi$ is locally given by multiplication, i.e., $\varphi(b \otimes b') = bb'$.
                  If $P$ is a point in $Y$, then there is an isomorphism $$(f_*\fO_X)_P \cong \fO_{P, X}[\varepsilon] / (\varepsilon^2 - 1) \cong \fO_{P, X} \times \fO_{P, X},$$
                  so $f$ is unramified at every $P \in Y$.
                  Hence, $f : X \to Y$ is a degree 2 étale cover of $Y$.

            \item[(c)] One direction is obvious.
                  Consider the pullback diagram
                  \[ \begin{tikzcd}
                      X \times_Y X \arrow[d] \arrow[r] & X \arrow[d] \\
                      X \arrow[r, "f"]                 & Y
                    \end{tikzcd}. \]
          \end{itemize}
          The projection morphisms $p_1, p_2 : X \times_Y X \to X$ are étale covers of degree 2.
          Also, the diagonal $\Delta \subset X \times_Y X$ is a closed subscheme of $X \times_Y X$ is isomorphic to $X$.
          Thus, the restriction $p_1, p_2 : X \times_Y X - \Delta \to X$ is an étale cover of $X$ of degree 1, which is an isomorphism.
          Hence, $\tau = p_2 \circ p_1^{-1} : X \to X$ is an automorphism of $X$ over $Y$ which tranposes the fibres of $f$.
          \filbreak

          Now, let $f : X \to Y$ be a étale cove of degree 2, and let $X' = \rSpec(\fO_Y \oplus \fL)$, where $\fL$ is the cokernel of the natural morphism $\fO_Y \to f_*\fO_X$.
          We want to show $X$ and $X'$ are isomorphic.
          It suffices to show the following short exact sequence of $\fO_Y$-modules splits
          \[ \begin{tikzcd}
              0 \arrow[r] & \fO_Y \arrow[r] & f_*\fO_X \arrow[r] & \fL \arrow[r] & 0.
            \end{tikzcd} \]
          The automorphism $\tau : X \to X$ of $X$ over $Y$, which tranposes the fibres of $f$, induces an $\fO_Y$-module automorphism $\tau_* : f_*\fO_X \to f_*\fO_X$.
          The kernel of the trace morphism $t = 1 - \tau : f_*\fO_X \to f_*\fO_X$ is exactly $\fO_Y$.
          Hence, $t$ induces a section $\fL \to f_*\fO_X$, so the short exact sequence splits.
        \end{proof}

\end{enumerate}
