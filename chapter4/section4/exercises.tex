      \item[\textbf{3.}] Let the elliptic curve $X$ be embedded in $\PP^2$ so as to have the equation $y^2 = x(x - 1)(x - \lambda)$.
            Show that any automorphism of $X$ leaving $P_0 = (0:1:0)$ fixed is induced by an automorphism of $\PP^2$ coming from the automorphism of the affine $(x, y)$-plane given by
            \begin{equation*}
                  \begin{cases*}
                        x' = ax + b \\
                        y' = cy.
                  \end{cases*}
            \end{equation*}
            In each of the four cases of (4.7), describe these automorphisms of $\PP^2$ explicitly, and hence determine the structure of the group $G = \Aut(X, P_0)$.

            \begin{proof}
                  The embedding of $X$ in $\PP^2$ is given by the complete linear system $|3P_0| \cong \PP^2$.
                  Let $1, x, y$ be a $k$-basis of $\cH^0(X, \fO_X(3P_0))$ such that $x, y$ are local affine coordinates of $|3P_0|$.
                  Any automorphism of $X$ leaving $P_0$ induces an automorphism of $|3P_0|$, i.e., there exists a faithful representation $\rho : G \hookrightarrow \Aut(|3P_0|) \cong\PGL(2)$.
                  If $\sigma \in G$, then we can write
                  \begin{equation*}
                        \rho(\sigma) = \begin{pmatrix}
                              a & 0 & b \\
                              c & d & e \\
                              f & 0 & g \\
                        \end{pmatrix}, \quad d(ag - bf) \neq 0.
                  \end{equation*}
                  The unique $g_2^1$, denoted $f : X \to \PP^1$, is given by the projection of $X$ onto the $x$-axis.
                  By (4.4), there exists $\tau \in \PGL(1)$ such that $\tau \circ f = f \circ \sigma$.

                  \fbox{$j \neq 0, 1728$} The unique nontrivial element $\sigma \in G$ interchanges the sheets of the unique $g_2^1$ and can be written as
                  \begin{equation*}
                        \begin{cases*}
                              x' = x \\
                              y' = -y
                        \end{cases*}.
                  \end{equation*}

                  \fbox{$j = 1728$ and $\chr{k} \neq 3$}

                  \fbox{$j = 0$ and $\chr{k} \neq 3$}

                  \fbox{$j = 0 ~ (= 1728)$ and $\chr{k} = 3$}
            \end{proof}

      \item[\textbf{4.}] Let $X$ be an elliptic curve in $\PP^2$ given by an equation of the form
            \begin{equation*}
                  y^2 + a_1xy + a_3y = x^3 + a_2x^2 + a_4x + a_6.
            \end{equation*}
            Show that the $j$-invariant is a rational function of the $a_i$, with coefficients in $\Q$.
            In particular, if the $a_i$ are all in some field $k_0 \subseteq k$, then $j \in k_0$ also.
            Furthermore, for every $\alpha \in k_0$, there exists an elliptic curve defined over $k_0$ with $j$-invariant equal to $\alpha$.

            \begin{proof}
                  Complete the square in $y$ to get an equation of the form
                  \begin{equation*}
                        y^2 = x^3 + Ax^2 + Bx + C,
                  \end{equation*}
                  with
                  \begin{equation*}
                        A = \frac{a_1^2}{4} + a_2, \quad B = \frac{a_1a_3}{2} + a_4, \quad C = \frac{a_3^2}{4} + a_6.
                  \end{equation*}
                  Let $\alpha, \beta, \gamma$ be the roots of the cubic on the right-hand side.
                  Make a linear change of $x$ to send $\alpha, \beta$ to $0, 1$ so that the equation becomes
                  \begin{equation*}
                        y^2 = x(x - 1)(x - \lambda), \quad \lambda = \frac{\gamma - \alpha}{\beta - \alpha}.
                  \end{equation*}
                  Plugging in $\lambda$ into the $j$-invariant formula gives
                  \begin{equation*}
                        j = -2^8 \frac{(\alpha + \beta + \gamma)^2 - 3(\alpha \beta - \beta \gamma - \gamma \alpha)}{(\alpha - \beta)^2(\beta - \gamma)^2(\gamma - \alpha)^2}.
                  \end{equation*}
                  This is a rational function of the symmetric polynomials in $\alpha, \beta, \gamma$, which are themselves polynomials in $A, B, C$, and hence in the $a_i$.
            \end{proof}

      \item[\textbf{13.}] If $p = 13$, there is just one value of $j$ for which the Hasse invariant of the corresponding curve is 0.
            Find it.

            \begin{proof}
                  The polynomial $h_p(\lambda)$ of (4.22) for $p = 13$ is
                  \begin{align*}
                        h_{13}(\lambda) = \lambda^6 + 10\lambda^5 + 4\lambda^4 + 10\lambda^3 + 4\lambda^2 + 10\lambda + 1,
                  \end{align*}
                  and an elliptic curve has Hasse invariant 0 if and only if $h_{13}(\lambda) = 0$.
                  On the otherhand, consider the $j$-invariant
                  \begin{equation*}
                        j = 2^8 \frac{(\lambda^2 - \lambda + 1)^3}{\lambda^2(\lambda - 1)^2}.
                  \end{equation*}
                  Expanding above modulo 13 gives
                  \begin{equation*}
                        \lambda^6 + 10\lambda^5 + (10j + 6)\lambda^4 + (6j + 6)\lambda^3 + (10j + 6)\lambda^2 + 10\lambda + 1 = 0.
                  \end{equation*}
                  Matching coefficients with $h_{13}(\lambda)$ and solving for $j$ gives $j \equiv 5 \mod{13}$.
            \end{proof}

      \item[\textbf{14.}] The Fermat curve $X : x^3 + y^3 = z^3$ gives a nonsingular curve in characteristic $p$ for every $p \neq 3$.
            Determine the set $\goth{P} = \{p \neq 3 \mid \text{$X_{(p)}$ has Hasse invariant 0}\}$, and observe (modulo Dirichlet's theorem) that it is a set of primes of density $1 / 2$.

            \begin{proof}
                  Using (4.21) with $f(x, y, z) = x^3 + y^3 - z^3$, we find that the coefficient of $(xyz)^{p - 1}$ in $f^{p - 1}$ is zero if and only if $p \equiv 2 \mod{3}$.
                  By Dirichlet's theorem, the density of $\goth{P}$ is $1 / \phi(3) = 1 / 2$.
            \end{proof}
