  \item Let $X$ be a curve, and let $P \in X$ be a point.
        Then there exists a nonconstant rational function $f \in K(X)$, which is regular everywhere except at $P$.

        \begin{proof}
          For large $n > 0$, we have $\cH^0(\fO_X(nP)) \neq 0$ (1.3.2).
          This implies the existence of an effective divisor $D'$ such that $D' - nP = (f)$, where $f$ is a rational function and $(f)$ is the divisor of zeros of $f$.
          Since $D'$ is effective, the only pole of $f$ is at $P$ of order at most $n$.
        \end{proof}

  \item[\textbf{5.}] For an effective divisor $D$ on a curve $X$ of genus $g$, show that $\dim{|D|} \leq \deg{D}$.
        Furthermore, equality holds if and only if $D = 0$ or $g = 0$.

        \begin{proof}
          The Riemann-Roch theorem gives
          \begin{equation*}
            \ell(D) - 1 = \ell(K - D) + \deg{D} - g,
          \end{equation*}
          so it suffices to show $\ell(K - D) \leq g$ and equality holds if and only if $D = 0$ or $g = 0$.

          If $D$ is effective, then we have an exact sequence
          \[\begin{tikzcd}
              0 \arrow[r] & \fO_X(-D) \arrow[r] & \fO_X \arrow[r] & \fO_D \arrow[r] & 0,
            \end{tikzcd} \]
          Tensoring by $\fO_X(K)$ and taking global sections gives an injection of vector spaces $\cH^0(\fO_X(K - D)) \to \cH^0(\fO_X(K))$, so we conclude that $\ell(K - D) \leq \ell(K) = g$.

          Now, we have $\dim|D| = \deg{D}$ if and only if $\ell(K - D) = \ell(K)$, and it is clear this equality holds if either $D = 0$ or $g = 0$.
          Conversely, suppose $D \neq 0$ and $g \neq 0$.
          We will show $\ell(K - D) < \ell(K)$.
          Since $D$ is effective, $d = \deg{D} > 0$, so we proceed by induction on $d$.
          If $D = P$ for a point $P \in X$, then $\ell(P) = 1$ because $X$ is not rational.
          Thus, Riemann-Roch gives $\ell(K - P) = g - 1$.

          Assume the statement for $d > 1$, and let $D$ be an effective divisor of degree $d + 1$.
          Let $P$ be a point in the support of $D$ so that $D' = D - P$ is an effective divisor of degree $d$.
          By the inductive hypothesis, we have $\ell(K - D') < \ell(K)$.
          Consider the exact sequence
          \[\begin{tikzcd}
              0 \arrow[r] & \fO_X(-P) \arrow[r] & \fO_X \arrow[r] & \fO_P \arrow[r] & 0.
            \end{tikzcd} \]
          Tensoring with $\fO_X(K - D')$ and taking global sections gives an inclusion of vector spaces $\cH^0(\fO_X(K - D)) \to \cH^0(\fO_X(K - D'))$.
          Thus, $\ell(K - D) \leq \ell(K - D')$, so combining inequalities, we get
          \begin{equation*}
            \ell(K - D) \leq \ell(K - D') < \ell(K).
          \end{equation*}
        \end{proof}

  \item[\textbf{6.}] Let $X$ be a curve of genus $g$.
        Show that there is a finite morphism $f : X \to \PP^1$ of degree $\leq g + 1$.

        \begin{proof}
          Fix a point $P \in X$, and consider the divisor $D = (g + 1)P$.
          By Riemann-Roch, $\ell(D) > 1$, which implies the existence of a nonconstant rational function $f \in K(X)$ with poles no worse than $D$.
          This means we can write $f = g / h$ such that the order of vanishing of $h$ at $P$ is at most $g + 1$, and $h$ has no other zeros.
          The inclusion of fields $k(f) \subseteq K(X)$ induces a finite morphism $f : X \to \PP^1$.
          Now, the degree of $f^{-1}(0)$ is at most $g + 1$ points, so $f$ has degree at most $g + 1$ by (II, 6.9).
        \end{proof}

  \item[\textbf{7.}] A curve $X$ is called \textit{hyperelliptic} if $g \geq 2$ and there exists a finite morphism $f : X \to \PP^1$ of degree 2.
        \begin{enumerate} [label=(\alph*)]
          \item If $X$ is a curve of genus $g = 2$, show that the canonical divisor defines a complete linear system $|K|$ of degree 2 and dimension 1, without basepoints. Use (II, 7.8.1) to conclude that $X$ is hyperelliptic.
          \item Show that the curves constructed in (1.1.1) all admit a morphism of degree 2 to $\PP^1$. Thus, there exist hyperelliptic curves of any genus $g \geq 2$.
        \end{enumerate}

        \begin{proof} $ $ \vspace{0pt}
          \begin{enumerate} [label=(\alph*), leftmargin=0cm]
            \item By Riemann-Roch, $\deg{K} = 2 g - 2 = 2$ and $\dim{|K|} = \ell(K) - 1 = 1$.
                  For any $P \in X$, consider the exact sequence
                  \[\begin{tikzcd}
                      0 \arrow[r] & \fO_X(K - P) \arrow[r] & \fO_X(K) \arrow[r] & \fO_P \arrow[r] & 0.
                    \end{tikzcd} \]
                  The linear system $|K|$ is basepoint free if and only if $\cH^0(\fO_X(K)) \to \cH^0(\fO_P)$ is surjective for every $P \in X$.
                  Taking the long exact sequence in cohomology gives
                  \[\begin{tikzcd}
                      0 \arrow[r] & \cH^0(\fO_X(K - P)) \arrow[r] & \cH^0(\fO_X(K)) \arrow[r] & \cH^0(\fO_P).
                    \end{tikzcd} \]
                  Since $\ell(K) = 2$ and $h^0(\fO_P) = 1$, it suffices to show $h^0(\fO_X(K - P)) = 1$.
                  Indeed, by the Riemann-Roch theorem, $\ell(K - P) = \ell(P)$, and $\ell(P) = 1$ because $X$ is not rational.
                  Hence, $\ell(K - P) = 1$.

            \item If $D$ is a divisor of type $(a, b)$ with $a, b > 0$, then $D$ is very ample, so by Bertini's theorem there exists an irreducible nonsingular curve $C \subseteq Q = \PP^1 \times \PP^1$ that is linearly equivalent to $D$ and has type $(a, b)$ as a divisor in $Q$.
            Also, $a = \deg \restr{p_2}{C}$ and $b = \deg \restr{p_1}{C}$.
            Suppose $(a, b) = (g + 1, 2)$.
            By (III, Ex. 5.6), $C$ has genus $=(g + 1 - 1)(2 - 1) = g$, and $\restr{p_1}{C} : C \to \PP^1$ is a morphism of degree 2.
          \end{enumerate}
        \end{proof}
