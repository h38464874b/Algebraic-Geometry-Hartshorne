\begin{enumerate} [label=\textbf{\arabic*.}, leftmargin=0em]

      \item The linear system of conics in $\P^2$ with two assigned base points $P_1$ and $P_2$ (4.1) determines a morphism $\psi$ of $X'$ (which is $\P^2$ with $P_1$ and $P_2$ blown up) to a nonsingular quadric surface $Y$ in $\P^3$, and furthermore $X'$ via $\psi$ is isomorphic to $Y$ with one point blown up.

            \begin{proof}
                  We have the commutative diagram
                  \[ \begin{tikzcd}
                              & X' \arrow[ld, "\pi"'] \arrow[rd, "\psi"] &                   &      \\
                              \P^2 \arrow[rr, "\varphi", dashed] &                                         & Y \arrow[r, hook] & \P^3
                        \end{tikzcd} , \]
                  where $\varphi$ is defined by the linear system $\gd$ of conics on $\P^2$ with assigned base-points $P_1$ and $P_2$, $Y$ is the image of $\varphi$, and $X'$ is the blow-up of $\P^2$ at $P_1$ and $P_2$.
                  We may take the two points to be $P_1 = (1 : 0 : 0)$ and $P_2 = (0 : 1 : 0)$.
                  Then the vector space $V \subseteq \cH^0(\P^2, \fO_{\P^2}(2))$ corresponding to $\gd$ is spanned by
                  \begin{equation*}
                        y_0 = x_1 x_2, \quad y_1 = x_0 x_2, \quad y_2  = x_0 x_1, \quad y_3 = x_2^2,
                  \end{equation*}
                  where $y_i$ are the homogenous coordinates of $\P^3$.
                  In particular, $\varphi$ is defined as
                  \begin{equation*}
                        \varphi(x_0 : x_1 : x_2) = (y_0 : y_1 : y_2: y_3),
                  \end{equation*}
                  and the points in the image of $\psi$ satisfies the equation
                  \begin{equation*}
                        y_0 y_1 - y_2 y_3 = 0,
                  \end{equation*}
                  which corresponds to a nonsingular quadric surface $Y \subset \P^3$.
                  It is a birational map since if we restrict to the affine patch defined by $x_2 \neq 0$, then we can identify $\varphi$ with a quadratic transformation (4.2.3).

                  For each $i = 1, 2$, let $E_i$ be the strict transform of $P_i$, let $M$ be the line through $P_1$ and $P_2$, which is defined by $x_2 = 0$ in $\P^2$, and let $\tilde{M}$ be the strict transform of $M$ in $X'$.
                  By (3.6), we have
                  \begin{equation*}
                        \tilde{M} \sim \pi^* M - E_1 - E_2,
                  \end{equation*}
                  so by (3.2), $\tilde{M}$ has self intersection
                  \begin{equation*}
                        \tilde{M}^2 = (M)^2 + (E_1)^2 + (E_2)^2 = -1.
                  \end{equation*}
                  Let $\lambda$ be a local coordinate on $M$ such that $M$ consists of the points $(\lambda : 1 : 0)$.
                  The image of $M$ under $\varphi$ for $\lambda \neq 0$ is computed to be
                  \begin{equation*}
                        \varphi(\lambda : 1 : 0) = (0 : 0 : \lambda : 0),
                  \end{equation*}
                  so denote by $Q$ the point $\varphi(M - \{ P_1, P_2 \}) = (0 : 0 : 1 : 0)$.
                  We claim $X'$ can be identified with the blowing-up of $Y$ at $Q$.
                  Indeed, $\varphi$ induces an isomorphism $$\P^2 - M \cong Y - \{ Q \} \cup \psi(E_1) \cup \psi(E_2) ,$$ and $\pi$ induces an isomorphism $$X' - E_1 \cup E_2 \cong \P^2 - \{ P_1, P_2 \},$$ so $\psi$ induces an isomorphism $$X' - \tilde{M} \cup E_1 \cup E_2 \cong Y - \{ Q \} \cup \psi(E_1) \cup \psi(E_2),$$
                  and since $\psi(E_i)$ are lines in $\P^3$ which intersect at $Q$, we conclude that $\psi$ induces an isomorphism
                  \begin{equation*}
                        X' - \tilde{M} \cong Y - \{ Q \}.
                  \end{equation*}
                  Hence by (5.7), $X'$ can be identified with $Y$ blown-up at $Q$.
                  (See [Manin, \SS 18] for more details.)
            \end{proof}

      \item Let $\varphi$ be the quadratic transformation of (4.2.3), centered at $P_1, P_2, P_3$.
            If $C$ is an irreducible curve of degree $d$ in $\P^2$, with points of multiplicity $r_1, r_2, r_3$ at $P_1, P_2, P_3$, then the strict transform $C'$ of $C$ by $\varphi$ has degree $d' = 2d - r_1 - r_2 - r_3$, and has points of multiplicity $d - r_2 - r_3$ at $Q_1$, $d - r_1 - r_3$ at $Q_2$, and $d - r_1 - r_2$ at $Q_3$.
            The curve may have arbitrary singularities.

            \begin{proof}
                  We have a commutative diagram
                  \[ \begin{tikzcd}
                              & X \arrow[ld, "\pi"'] \arrow[rd, "\psi"] &      \\
                              \P^2 \arrow[rr, "\varphi", dashed] &                                          & \P^2
                        \end{tikzcd} . \]
                  Let $H$ be a hyperplane section of $\P^2$, so that $\varphi$ is given by the linear system $\gd = |2H - P_1 - P_2 - P_3|$, and let $E_1, E_2, E_3$ be the exceptional divisors of $X$ via the blowing-up of $\P^2$ at $P_1, P_2, P_3$, respectively.
                  By (3.6), the strict transform $\tilde{C}$ of $C$ by $\pi$ is linearly equivalent to
                  \begin{equation*}
                        \tilde{C} = \pi^*C - r_1E_1 - r_2 E_2 - r_3 E_3.
                  \end{equation*}
                  Since $\psi$ is given by the linear system $\gd' = |D|$ with $D = 2\pi^* H - E_1 - E_2 - E_3$, $\varphi$ is a birational map such that $C' = \varphi(C) = \psi(\tilde{C})$, and $C \sim dH$ in $\P^2$, the degree of $C'$ in $\P^2$ can be computed using (Ex. 1.2) and (3.2)
                  \begin{align*}
                        \tilde{C}.D & = (d\pi^*H - r_1E_1 - r_2E_2 - r_3 E_3). (2\pi^*H - E_1 - E_2 - E_3) \\
                                    & = 2d(H^2) + r_1 (E_1)^2 + r_2 (E_2)^2 + r_3(E_3)^2                   \\
                                    & = 2d - r_1 - r_2 - r_3.
                  \end{align*}
                  For each $i \neq j$, let $L_{ij}$ be the line which goes through $P_i$ and $P_j$.
                  By (3.6), for each $k = 1, 2, 3$, the multiplicity of $C'$ at $Q_k$ is the intersection of $\tilde{C}$ with the strict transform $\tilde{L}_{ij}$ of $L_{ij}$ with $i, j \neq k$ via $\pi$ (4.2.3), which by (3.6) is linearly equivalent to
                  \begin{equation*}
                        \tilde{L}_{ij} = \pi^*L_{ij} - E_i - E_j.
                  \end{equation*}
                  Again, we have $L_{ij} \sim H$ in $\P^2$, so
                  \begin{align*}
                        \tilde{C}.\tilde{L}_{ij} & = (d\pi^*H - r_1E_1 - r_2 E_2 - r_3E_3).(\pi^*H - E_i - E_j) \\
                                                 & = d(H)^2 + r_i (E_i)^2 + r_j (E_j)^2                         \\
                                                 & = d - r_i - r_j.
                  \end{align*}
                  Hence, $C'$ has points of multiplicity $d - r_i - r_j$ at $Q_k$.
            \end{proof}

      \item Let $C$ be an irreducible curve in $\P^2$.
            Then there exists a finite sequence of quadratic transformations, centered at suitable triples of points, so that the strict transform $C'$ of $C$ has only \textit{ordinary} singularities, i.e., multiple points with all distinct tangent directions.

            \begin{proof}
                  If $C$ is a conic in $\P^2$, then it is nonsingular, so assume $C$ has degree $\geq 3$, and let $P_1$ be a non-ordinary singular point of $C$.
                  Our goal is to find points $P_2, P_3 \in \P^2$ such that the quadratic transformation centered at $P_i$ improves the singularity at $P_1$ and only introduces ordinary singularities on the strict transform of $C$.
                  To that end, for each $i \neq j$, let $L_{ij}$ be the line through $P_i$ and $P_j$.
                  The restriction of the quadratic transformation $\varphi : \P^2 \to \P^2$ centered at $P_i$ to $\P^2 - \bigcup L_{ij}$ is an isomorphism onto its image, and each $L_{ij}$ is sent to a point $Q_k$ with $k \neq i, j$ such that if $M$ is a curve meeting $L_{ij}$ transversally away from the centers, then the strict transform of $M$ has ordinary singularities at $Q_k$.
                  Therefore, we require that away from $P_1$, the lines $L_{ij}$ meets $C$ transversally, and $L_{12}, L_{13}$ are distinct from the tangent directions of $C$ at $P$.
                  There are only finitely many tangent directions of $C$ at $P_1$, and since $C$ is not strange (IV, 3.8) (IV, 3.9), we can find many lines $L_{12}$ and $L_{13}$ through $P_1$ with tangent vectors distinct from that of $C$ at $P_1$ which meets $C$ transversally.
                  Lastly, pick $L_{23}$ which avoids $P_1$, not parallel to $L_{12}, L_{13}$, meets $C$ transversally, and the points $P_2, P_3$ of intersection of $L_{23}$ with $L_{12}, L_{13}$, respectively, are not on $C$, which again exists because $C$ is not strange.
                  The strict transform of $C$ under the quadratic transformation centered at $P_i$ improves the singularity at $P_1$ (3.7) and only introduces ordinary singularities at $Q_k$.
                  By (3.9), we can repeat this process for each nonordinary singularity until the strict transform only has ordinary singularities introduced by the quadratic transformations.
            \end{proof}

      \item \begin{itemize}
                  \item[(a)] Use (4.5) to prove the following lemma on cubics: If $C$ is an irreducible plane cubic curve, if $L$ is a line meeting $C$ in points $P, Q, R$, and $L'$ is a line meeting $C$ in points $P', Q', R'$, let $P''$ be the third intersection of the line $PP'$ with $C$, and define $Q''$, $R''$ similarly.
                        Then $P'', Q'', R''$ are collinear.

                  \item[(b)] Let $P_0$ be an inflection point of $C$, and \textit{define} the group operation on the set of regular points of $C$ by the geometric recipe ``let the line $PQ$ meet $C$ at $R$, and let $P_0R$ meet $C$ at $T$, then $P + Q = T$'' as in (II, 6.10.2) and (II, 6.11.4).
                        Use (a) to show that this operation is associative.
            \end{itemize}

            \begin{proof} $ $ \vspace{0pt}
                  \begin{itemize} [leftmargin=0cm]
                        \item[(a)] Let $L_P$ be the line $PP'$, and similarly for $L_Q, L_R$, and let $M$ be the line going through $P''$ and $Q''$.
                              The divisors $C$ and $L_P + L_Q + L_R$ belong to the linear system of conics in $\P^2$ with assigned base points $P, P', P'', Q, Q'', Q''', R, R'$, and they intersect at $R''$, so by (4.5) $R''$ is the unique point in $\P^2$ such that every cubic curve in $\P^2$ going through $P, P', P'', Q, Q'', Q''', R, R'$ also goes through $R''$.
                              In particular, $L + L' + M$ belongs to this linear system, so it must go through $R''$.
                              Therefore, $M$ must go through $R''$.

                        \item[(b)] Let $P_1, P_2, P_3$ be regular points of $C$, let $L_{12}$ be the line $P_1P_2$, let $R_{12}$ be the third point $L_{12}$ meets $C$, let $L'_{12}$ be the line $P_0 R_{12}$, let $Q_{12} = P_1 + P_2$, and similarly for $L_{23}, L'_{23}$, and $R_{23}, Q_{23}$.
                              Let $M_{12, 3}$ be the line $Q_{12}P_{3}$, let $S_{12, 3}$ be the third point $M_{12, 3}$ meets $C$, let $M'_{12, 3}$ be the line $P_0S_{12, 3}$, let $T_{12, 3} = Q_{12} + P_3$, and similarly for $M_{1, 23}, M'_{1, 23}$, and $S_{1, 23}, T_{1, 23}$.
                              We want to show $T_{12, 3} = T_{1, 23}$.
                              Applying (a) to
                              \begin{equation*}
                                    L = L_{12}, \quad L' = L'_{23},
                              \end{equation*}
                              and
                              \[ \begin{aligned}
                                          PP' & = M_{1,23}, & P'' & = S_{1,23}, \\
                                          QQ' & = L_{23},   & Q'' & = P_3,      \\
                                          RR' & = L'_{12},  & R'' & = Q_{12},
                                    \end{aligned} \]
                              shows $S_{1, 23}$, $P_3$, and $Q_{12}$ are collinear.
                              Therefore, $S_{1, 23} = S_{12, 3}$, and so $T_{1, 23} = T_{12, 3}$.
                  \end{itemize}
            \end{proof}

      \item If $D$ is any divisor of degree $d$ on the cubic surface (4.7.3), show that
            \begin{equation*}
                  p_a(D) \leq \begin{cases*}
                        \cfrac{1}{6}(d - 1)(d - 2)                & \text{if $d \equiv 1, 2 \mod{3}$} \\

            \end{equation*}

      \item Show that a divisor class $D$ on the cubic surface contains an irreducible curve $\iff$ it contains an irreducible nonsingular curve $\iff$ it is either (a) one of the 27 lines, or (b) a conic (meaning a curve of degree 2) with $D^2 = 0$, or (c) $D.L \geq 0$ for every line $L$, and $D^2 > 0$.

      \item If $C$ is an irreducible non-singular curve of degree $d$ on the cubic surface, and if the genus $g > 0$, then

      \item A curious consequence of the implications (iv) $\implies$ (iii) of (4.11) is the following numerical fact: Given integers $a, b_1, \dots, b_6$ such that $b_i > 0$ for each $i$, $a - b_i - b_j > 0$ for each $i, j$, and $2a - \sum_{i \neq j} b_i > 0$ for each $j$, we must necessarily have $a^2 - \sum b_i^2 > 0$.
            Prove this directly (for $a, b_1, \dots, b_6 \in \R$) using methods of freshman calculus.

      \item \textit{The Weyl Groups.} Given any diagram consisting of points and line segments joining some of them, we define an abstract group, given by the generators and relations, as follows: each point represents a generator $x_i$.
            The relations are $x_i^2 = 1$ for each $i$; $(x_ix_j)^2 = 1$ if $i$ and $j$ are not joined by a line segment, and $(x_ix_j)^3 = 1$ if $i$ and $j$ are joined by a line segment.
            \begin{itemize}
                  \item[(a)] The \textit{Weyl group} $\mathbf{A}_n$ is defined using the diagram
                        \tble{A/{}}
                        of $n - 1$ points, each joined to the next.
                        Show that it is isomorphic to the symmetric group $\Sigma_n$ as follows: map the generators of $\A_n$ to the elements $(12), (23), \dots, (n - 1, n)$ of $\Sigma_n$, to get a surjective homomorphism $\qA_n \to \Sigma_n$.
                        Then estimate the number of elements of $\qA_n$ to show in fact it is an isomorphism.
                  \item[(b)] The \textit{Weyl group} $\mathbf{E}_6$ is defined using the diagram
                        \tble{E/{6}}
                        Call the generators $x_1, \dots, x_5$ and $y$.
                        Show that one obtains a surjective homomorphism $\mathbf{E}_6 \to G$, the group of automorphisms of the configuration of 27 lines (4.10.1), by sending $x_1, \dots, x_5$ to the permutations $(12), (23), \dots, (56)$ of the $E_i$, respectively, and $y$ to the element associated with the quadratic transformation based at $P_1, P_2, P_3$.
                  \item[(c)] Estimate the number of elements in $\mathbf{E}_6$, and thus conclude that $\mathbf{E}_6 \cong G$.
            \end{itemize}

      \item Let $P_1, \dots, P_4$ be a fintie set of (ordinary) points of $\P^2$, no 3 collinear.
            We define an \textit{admissible transformation} to be a quadratic transformation (4.2.3) centered at some three of the $P_i$ (call them $P_1, P_2, P_3$).
            This gives a new $\P^2$, and a new set of $r$ points, namely $Q_1, Q_2, Q_3$, and the images of $P_4, \dots, P_r$.
            We say that $P_1, \dots, P_r$ are in \textit{general position} if no three are collinear, and furthermore after any finite sequence of admissible transformations, the new set of $r$ points also have no three collinear.
            \begin{itemize}
                  \item[(a)] A set of 6 points is in general position if and only if no three are collinear and not all six lie on a conic.
                  \item[(b)] If $P_1, \dots, P_r$ are in general position, then the $r$ points obtained by any finite sequence of admissible transformations are also in general position.
                  \item[(c)] Assume the ground field $k$ is uncountable.
                        Then given $P_1, \dots, P_r$ in general position, there is a dense $V \subseteq \P^2$ such that for any $P_{r + 1} \in V$, $P_1, \dots, P_{r + 1}$ will be in general position.
                  \item[(d)] Now take $P_1, \dots, P_r \in \P^2$ in general position, and let $X$ be the surface obtained by blowing up $P_1, \dots, P_r$.
                        If $r = 7$, show that $X$ has exactly 56 irreducible nonsingular curves $C$ with $g = 0$, $C^2 = -1$, and that these are the only irreducible curves with negative self-intersection.
                        Ditto for $r = 8$, the number being 240.
                  \item[(e)] For $r = 9$, show that the surface $X$ defined in (d) has infinitely many irreducible nonsingular curves $C$ with $g = 0$ and $C^2 = -1$.
            \end{itemize}
