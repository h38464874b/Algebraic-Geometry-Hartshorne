    \item Let $X$ be a nonsingular projective variety of any dimension, let $Y$ be a nonsingular subvariety, and let $\pi : \tilde{X} \to X$ be obtained by blowing up $Y$.
          Show that $p_a(\tilde{X}) = p_a(X)$.

          \begin{proof}
              The proof of (3.4) follows through for the more general case by (II, 8.24).

              \begin{proposition}
                  In the above setting, we have $\pi_*\fO_{\tilde{X}} = \fO_X$, and $\dR^i \pi_*\fO_{\tilde{X}} = 0$ for $i > 0$.
                  Therefore, $\cH^i(X, \fO_X) \cong \cH^i(\tilde{X}, \fO_{\tilde{X}})$ for all $i \geq 0$.
              \end{proposition}
          \end{proof}

    \item[\textbf{5.}] Let $a_1, \dots, a_r$, $r \geq 5$, be distinct elements of $k$, and let $C$ be the curve in $\P^2$ given by the (affine) equation $y^2 = \prod_{i = 1}^r (x - a_i)$.
          Show that the point $P$ at infinity on the $y$-axis is a singular point.
          Compute $\delta_P$ and $g(\tilde{Y})$, where $\tilde{Y}$ is the normalization of $Y$.
          Show in this way that one obtains hyperelliptic curves of genus $g \geq 2$.

          \begin{proof}
              The point $P$ at infinity on the $y$-axis has homogenous coordinates $(0 : 1: 0)$, and the local equation of $Y$ at $P$ at the affine patch defined by $y \neq 0$ is given by $z^{r - 2} = \prod_{i = 1}^r (x - a_i z)$, so $Y$ has multiplicity $\mu_P(Y) = r - 2$.
              Let us see what happens when we blow up $\P^2$ at $P$.
              The substitution $z = ux$ gives
              \begin{equation*}
                  x^{r - 2}\bigg(u^{r - 2} - x^2 \prod_{i = 1}^r (1 - a_i u) \bigg) = 0,
              \end{equation*}
              so the strict transform $Y_1$ of $Y$ is
              \begin{equation*}
                  u^{r - 2} - x^2 \prod_{i = 1}^r (1 - a_i u) = 0.
              \end{equation*}
              Since $Y_1$ is still singular at $(0, 0)$, we blow up $Y_1$ by making the substitution $x = vu$, which gives
              \begin{equation*}
                  u^2\bigg( u^{r - 4} - v^2 \prod_{i = 1}^r (1 - a_i u)\bigg) = 0,
              \end{equation*}
              so the strict transform $Y_2$ of $Y_1$ is
              \begin{equation*}
                  u^{r - 4} - v^2 \prod_{i = 1}^r (1 - a_i u) = 0.
              \end{equation*}
              We can continue this process until the exponent of $u$ is either 0 or 1, which depends on whether $r$ is even or odd.
              In the former case, $r$ is even, and the equation of the strict transform of $Y$ after $r/2$ blow-ups is given by
              \begin{equation*}
                  1 - w^2 \prod_{i = 1}^r (1 - a_i u) = 0,
              \end{equation*}
              which is nonsingular.
              The multiplicities at the points that were blown up were $r - 2, 2, \dots, 2$, so the genus of $\tilde{Y}$ can be computed using (3.9.2)
              \begin{equation*}
                  g(\tilde{Y}) = \frac{(r-1)(r-2)}{2} -\frac{(r-2)(r-3)}{2} - \frac{r - 4}{2} =\boxed{\frac{r-2}{2}}.
              \end{equation*}
              If $r$ is odd, then the strict transform of $Y$ after $(r - 1)/2$ blow-ups is given by
              \begin{equation*}
                  u - w^2 \prod_{i = 1}^r (1 - a_i u) = 0,
              \end{equation*}
              which is also nonsingular.
              Similar to the case $r$ is even, the genus of $\tilde{Y}$ can be computed using (3.9.2)
              \begin{equation*}
                  g(\widetilde Y) =\frac{(r-1)(r-2)}2 -\frac{(r-2)(r-3)}2 -\frac{r-3}{2} =\boxed{\frac{r-1}{2}}.
              \end{equation*}
              Composing the projection of $Y \to \P^1$ onto the $x$-axis with the normalization $\tilde{Y} \to Y$ is a $g^1_2$ on $\tilde{Y}$, so $\tilde{Y}$ is a hyperelliptic curve.
          \end{proof}

    \item[\textbf{7.}] For each of the following singularities at $(0, 0)$ in the plane, give an embedded resolution, compute $\delta_P$, and decide which ones are equivalent.
          \begin{itemize}
              \item[(a)] $x^3 + y^5 = 0$
              \item[(b)] $x^3 + x^4 + y^5 = 0$
              \item[(c)] $x^3 + y^4 + y^5 = 0$
              \item[(d)] $x^3 + y^5 + y^6 = 0$
              \item[(e)] $x^3 + xy^3 + y^5 = 0$
          \end{itemize}
          \begin{proof} $ $ \vspace{0pt}
              \begin{itemize} [leftmargin=0cm]
                  \item[(a)] The substitution $x = uy$ gives
                        \begin{equation*}
                            y^3(u^3 + y^2) = 0,
                        \end{equation*}
                        so after the first blow-up, we obtain a cuspidal curve. Thus,
                        \begin{equation*}
                            E_1 : y = 0, \quad C_1 : u^3 + y^2 = 0,
                        \end{equation*}
                        where $E_1$ is the exceptional locus and $C_1$ is the strict transform of $C$.
                        Next, the substitution $y = uv$ gives
                        \begin{equation*}
                            u^2(u + v^2) = 0,
                        \end{equation*}
                        so the second blow-up gives
                        \begin{equation*}
                            E_2 : u = 0, \quad C_2 : u + v^2 = 0.
                        \end{equation*}
                        The substitution $u = vw$ then gives
                        \begin{equation*}
                            v(w + v) = 0,
                        \end{equation*}
                        so the third blow-up gives
                        \begin{equation*}
                            E_3 : v = 0, \quad C_3 : w + v = 0.
                        \end{equation*}
                        Finally, the substitution $v = wt$ gives
                        \begin{equation*}
                            w(t + 1) = 0,
                        \end{equation*}
                        so the fourth blow-up gives
                        \begin{equation*}
                            E_4 : w = 0, \quad C_4 : t + 1 = 0.
                        \end{equation*}
                        The total transform of the original curve is
                        \begin{equation*}
                            w^{15} t^9 (t + 1) = 0,
                        \end{equation*}
                        and we compute $\delta_P$ using (3.9.3):
                        \begin{equation*}
                            \delta_P = \frac{1}{2}3(3 - 1) + \frac{1}{2}2(2 - 1) + \frac{1}{2}1(1 - 1) + \frac{1}{2}1(1 - 1) = 4.
                        \end{equation*}

                  \item[(b)] The substitution $x = uy$ gives
                        \begin{equation*}
                            y^3(u^3 + u^4 y + y^2) = 0,
                        \end{equation*}
                        so the first blow-up gives
                        \begin{equation*}
                            E_1 : y = 0, \quad C_1 : u^3 + u^4 y + y^2 = 0.
                        \end{equation*}
                        Next, the substitution $y = uv$ gives
                        \begin{equation*}
                            u^2(u + u^3 v + v^2) = 0,
                        \end{equation*}
                        so the second blow-up gives
                        \begin{equation*}
                            E_2 : u = 0, \quad C_2 = u + u^3 v + v^2 = 0.
                        \end{equation*}
                        The substitution $u = vw$ then gives
                        \begin{equation*}
                            v(w + v^3w^3 + v) = 0,
                        \end{equation*}
                        so the third blow-up gives
                        \begin{equation*}
                            E_3 : v = 0, \quad C_3 : w + v^3 w^3 + v = 0.
                        \end{equation*}
                        Finally, the substitution $v = wt$ gives
                        \begin{equation*}
                            w(1 + w^5t^3 + t) = 0,
                        \end{equation*}
                        so the fourth blow-up gives
                        \begin{equation*}
                            E_4 : w = 0, \quad C_4 : 1 + w^5 t^3 + t = 0.
                        \end{equation*}
                        The total transform of the original curve is
                        \begin{equation*}
                            w^{15}t^9(1 + w^5 t^3 + t) = 0,
                        \end{equation*}
                        and we compute $\delta_P$ using (3.9.3):
                        \begin{equation*}
                            \delta_P = \frac{1}{2}3(3 - 1) + \frac{1}{2}2(2 - 1) + \frac{1}{2}1(1 - 1) + \frac{1}{2}1(1 - 1) = 4.
                        \end{equation*}

                  \item[(c)] The substitution $x = uy$ gives
                        \begin{equation*}
                            y^3(u^3 + y + y^2) = 0,
                        \end{equation*}
                        so the first blow-up gives
                        \begin{equation*}
                            E_1 : y = 0, \quad C_1 : u^3 + y + y^2 = 0.
                        \end{equation*}
                        Next, the substitution $y = uv$ gives
                        \begin{equation*}
                            u(u^2 + v + uv^2) = 0,
                        \end{equation*}
                        so the second blow-up gives
                        \begin{equation*}
                            E_2 : u = 0, \quad C_2 : u^2 + v + uv^2 = 0.
                        \end{equation*}
                        The substitution $v = uw$ then gives
                        \begin{equation*}
                            u(u + w + u^2w^2) = 0,
                        \end{equation*}
                        so the third blow-up gives
                        \begin{equation*}
                            E_3 : u = 0, \quad C_3 : u + w + u^2w^2 = 0.
                        \end{equation*}
                        Finally, the substitution $u = wt$ gives
                        \begin{equation*}
                            w(t + 1 + w^3t^2) = 0,
                        \end{equation*}
                        so the fourth blow-up gives
                        \begin{equation*}
                            E_4 : w = 0, \quad C_4 : t + 1 + w^3t^2 = 0.
                        \end{equation*}
                        The total transform of the original curve is
                        \begin{equation*}
                            w^{12}t^8(t + 1 + w^3t^2) = 0,
                        \end{equation*}
                        and we compute $\delta_P$ using (3.9.3):
                        \begin{equation*}
                            \delta_P = \frac{1}{2}3(3 - 1) + \frac{1}{2}1(1 - 1) + \frac{1}{2}1(1 - 1) + \frac{1}{2}1(1 - 1) = 3.
                        \end{equation*}

                  \item[(d)] The substitution $x = uy$ gives
                        \begin{equation*}
                            y^3(u^3 + y^2 + y^3) = 0,
                        \end{equation*}
                        so the first blow-up gives
                        \begin{equation*}
                            E_1 : y = 0, \quad C_1 : u^3 + y^2 + y^3 = 0.
                        \end{equation*}
                        Next, the substitution $y = uv$ gives
                        \begin{equation*}
                            u^2(u + v^2 + uv^3) = 0,
                        \end{equation*}
                        so the second blow-up gives
                        \begin{equation*}
                            E_2 : u = 0, \quad C_2 : u + v^2 + uv^3 = 0.
                        \end{equation*}
                        The substitution $u = vw$ then gives
                        \begin{equation*}
                            v(w + v + v^3 w) = 0,
                        \end{equation*}
                        so the third blow-up gives
                        \begin{equation*}
                            E_3 : v = 0, \quad C_3 : w + v + v^3w = 0.
                        \end{equation*}
                        Finally, the substitution $v = wt$ gives
                        \begin{equation*}
                            w(1 + t + w^3t^3) = 0,
                        \end{equation*}
                        so the fourth blow-up gives
                        \begin{equation*}
                            E_4 : w = 0, \quad C_4 : 1 + t + w^3t^3 = 0.
                        \end{equation*}
                        The total transform of the original curve is
                        \begin{equation*}
                            w^{15}t^9(1 + t + w^3t^3) = 0,
                        \end{equation*}
                        and we compute $\delta_P$ using (3.9.3):
                        \begin{equation*}
                            \delta_P = \frac{1}{2}3(3 - 1) + \frac{1}{2}2(2 - 1) + \frac{1}{2}1(1 - 1) + \frac{1}{2}1(1 - 1) = 4.
                        \end{equation*}

                  \item[(e)] The substitution $x = uy$ gives
                        \begin{equation*}
                            y^3(u^3 + uy + y^2) = 0,
                        \end{equation*}
                        so the first blow-up gives
                        \begin{equation*}
                            E_1 : y = 0, \quad C_1 : u^3 + uy + y^2 = 0.
                        \end{equation*}
                        Next, the substitution $y = uv$ gives
                        \begin{equation*}
                            u^2(u + v + v^2) = 0,
                        \end{equation*}
                        so the second blow-up gives
                        \begin{equation*}
                            E_2 : u = 0, \quad C_2 : u + v + v^2 = 0.
                        \end{equation*}
                        The substitution $u = vw$ then gives
                        \begin{equation*}
                            v(w + 1 + v) = 0,
                        \end{equation*}
                        so the third blow-up gives
                        \begin{equation*}
                            E_3 : v = 0, \quad C_3 : w + 1 + v = 0.
                        \end{equation*}
                        The total transform of the original curve is
                        \begin{equation*}
                            v^9 w^5 (vw)^2(v^2w)^3(w + 1 + v) = 0,
                        \end{equation*}
                        and we compute $\delta_P$ using (3.9.3):
                        \begin{equation*}
                            \delta_P = \frac{1}{2}3(3 - 1) + \frac{1}{2}2(2 - 1) + \frac{1}{2}1(1 - 1) = 4.
                        \end{equation*}
              \end{itemize}

              The singularities in (a), (b), and (d) are equivalent, while the singularities in (c) and (e) are not equivalent to any of the others.
          \end{proof}

\end{enumerate}
