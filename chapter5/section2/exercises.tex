    \item If $X$ is a birationally ruled surface, show that the curve $C$, such that $X$ is birationally equivalent to $C \times \P^1$, is unique up to isomorphism.

          \begin{proof}
              Let $C$ and $C'$ be two nonsingular curves such that $X$ is birational to $C \times \P^1$ and $C' \times \P^1$.
              Then, there exists rational maps $\pi, \pi' : X \dasharrow C, C'$ and sections $\sigma, \sigma' : C, C' \to X$ such that $\pi \circ \sigma$ and $\pi' \circ \sigma'$ extend to the identity morphism on $C$ and $C'$, respectively.
              Then $\pi' \circ \sigma : C \to C'$ and $\pi \circ \sigma' : C' \to C$ are inverses to each other, so we have an isomorphism $C \cong C'$.
          \end{proof}

    \item Let $X$ be the ruled surface $\P(\fE)$ over a curve $C$.
          Show that $\fE$ is decomposable if and only if there exist two sections $C', C''$ of $X$ such that $C' \cap C'' = \emptyset$.

          \begin{proof}
              If $\fE$ is decomposable, then $\fE = \fO \oplus \fL$, where $\fL = \fO_C(\ge)$ for some divisor $\ge \in \Div{C}$.
              By (2.8.1), (2.11.1), and (2.11.2), the two natural projections $\fE \to \fL$ and $\fE \to \fO_C$ correspond to sections $C' \sim C_0$ and $C'' \sim C_0 - \ge f$, respectively, such that $C'.C'' = 0$.
              Hence, $C' \cap C'' = \emptyset$.
              Conversely, suppose there exist two sections $C', C''$ of $X$ such that $C' \cap C'' = \emptyset$.
              The corresponding surjective morphisms $\fE \to \fL'$ and $\fE \to \fL''$ induce an $\fO_C$-module homomorphism $\fE \to \fL' \oplus \fL''$.
              For every $P \in C$, the morphism of stalks $\fE_P \to \fL'_P \oplus \fL''_P$ is an isomorphism because $C'$ and $C''$ are disjoint, so $\fL'_P \oplus \fL''_P$ is generated by $\fE_P$.
              Hence, $\fE \to \fL' \oplus \fL''$ is an isomorphism.
          \end{proof}

    \item \begin{itemize}
              \item[(a)] If $\fE$ is a locally free sheaf of rank $r$ on a nonsingular curve $C$, then there is a sequence
                    \begin{equation*}
                        0 = \fE_0 \subseteq \fE_1 \subseteq \cdots \subseteq \fE_r = \fE
                    \end{equation*}
                    of subsheaves such that $\fE_i / \fE_{i - 1}$ is an invertible sheaf for each $i = 1, \dots, r$.
                    We say that $\fE$ is a \textit{successive extension} of invertible sheaves.
              \item[(b)] Show that this is false for varieties of dimension $\geq 2$.
                    In particular, the sheaf of differentials $\Omega_{\P^2}$ on $\P^2$ is not an extension of invertible sheaves.
          \end{itemize}

          \begin{proof} $ $ \vspace{0pt}
              \begin{itemize} [leftmargin=0cm]
                  \item[(a)] We proceed by induction on $r$.
                        The base case $r = 1$ is trivial, so assume the statement for any $r > 1$.
                        We can twist $\fE$ by a very ample sheaf $\fL$ so that $\fE' = \fE \otimes \fE$ is generated by global sections.
                        If we prove the statement for $\fE'$ so we have a filtration $(\fE'_i)$, then tensoring by $\fL^{-1}$ gives a filtration $(\fE_i = \fE'_i \otimes \fL^{-1})$ of $\fE$ such that $$\fE_i / \fE_{i - 1} \cong (\fE_i' / \fE'_{i - 1}) \otimes \fL^{-1},$$
                        which is an invertible sheaf for each $i = 1, \dots, r$.
                        Hence, we can assume $\fE$ is generated by global sections.
                        The hypothesis of (II, Ex. 8.2) is satisfied for $\fE$, so there exists a short exact
                        \[ \begin{tikzcd}
                                0 \arrow[r] & \fO_C \arrow[r] & \fE \arrow[r] & \fF \arrow[r] & 0,
                            \end{tikzcd} \]
                        where $\fF$ is also globally generated locally free of rank $r - 1$.
                        By the inductive hypothesis, this means we can find a filtration $(\fF_i)$ of $\fF$ such that each successive quotient is an invertible sheaf.
                        Consider the preimage $(\fE_j)$ of $(\fF_i)$, where we relabel $j = i + 1$.
                        Each $\fE_j$ is also locally free, $\fE_1 = \im(\fO_C)$ by exactness, and there is a canonical isomorphism $\fE_j / \fE_{j - 1} \cong \fF_{j - 1} / \fF_{j - 2}$.
                        Extending $(\fE_j)$ with $\fE_0 = 0$ gives a successive extension of sheaves.

                  \item[(b)] Suppose we have an exact sequence
                        \[ \begin{tikzcd}
                                0 \arrow[r] & \fL \arrow[r] & \Omega \arrow[r] & \fM \arrow[r] & 0,
                            \end{tikzcd} \]
                        where $\fL$ and $\fM$ are invertible sheaves on $\P^2$.
                        Then $\cH^1(\fL), \cH^1(\fM) = 0$, so by the long exact sequence in cohomology, we have $\cH^1(\Omega) = 0$.
                        But by (III, Ex. 7.2), $\cH^1(\Omega) = k$, a contradiction.
              \end{itemize}
          \end{proof}

    \item Let $C$ be a curve of genus $g$, and let $X$ be the ruled surface $C \times \P^1$.
          We consider the question, for what integers $s \in \Z$ does there exist a section $D$ of $X$ with $D^2 = s$?
          First show that $s$ is always an even integer, say $s = 2r$.
          \begin{itemize}
              \item[(a)] Show that $r = 0$ and $r \geq g + 1$ are always possible.
              \item[(b)] If $g = 3$, show that $r = 1$ is not possible, and just one of the two values $r = 2, 3$ is possible, depending on whether $C$ is hyperelliptic or not.
          \end{itemize}

          \begin{proof}
              Fix a section $C_0 = C \times P_0 \subset X$, where $P_0$ is a point of $\P^1$.
              The ruled surface $X = C \times \P^1$ corresponds to the projective bundle of $\fE = \fO_C \oplus \fO_C$.
              The invariant of $X$ is $e = 0$, which means $C_0^2 = -e = 0$.
              By (2.9), if $D$ is any section of $X$ corresponding to a surjection $\fE \to \fO_C(\gd)$ for some divisor $\gd$ of $C$ with $r = \deg \gd$, then $r = C_0.D$ and $D \sim C_0 + \gd f$.
              Hence, $s = 2r$.

              \begin{itemize} [leftmargin=0cm]
                  \item[(a)] There is a one-to-one correspondence between sections $D$ of $X$ and finite separable morphisms $f : C \to \P^1$.
                        In particular, a section $D$ is the graph of a morphism $f : C \to \P^1$, and its self-intersection is $D^2 = 2 \cdot \deg{f}$.
                        By (IV, Ex. 1.6), $g + 1$ is an upperbound on the minimum of $\deg{f}$.
                        Hence, $r = 0$ and $r \geq g + 1$ is always possible.

                  \item[(b)] If $g > 0$, then there is no morphism $C \to \P^1$ of degree 1.
                        If $C$ is a hyperelliptic curve, then it has a $g^1_2$, so $r = 2, 3$ is possible.
                        Otherwise, nonhyperelliptic curves of genus 3 always have a $g^1_3$ (IV, 5.5.2), so $r = 3$ is always possible.
              \end{itemize}
          \end{proof}

    \item \textit{Values of $e$.} Let $C$ be a curve of genus $g \geq 1$.
          \begin{itemize}
              \item[(a)] Show that for each $0 \leq e \leq 2g - 2$ there is a ruled suface $X$ over $C$ with invariant $e$, corresponding to an indecomposable $\fE$.
              \item[(b)] Let $e < 0$, let $D$ be any divisor of degree $d = -e$, and let $\xi \in \cH^1(X, \fO(-D))$ be a nonzero element defining an extension
                    \[ \begin{tikzcd}
                            0 \arrow[r] & \fO \arrow[r] & \fE \arrow[r] & \fO(D) \arrow[r] & 0.
                        \end{tikzcd} \]
                    Let $H \subseteq |D + K|$ be the sublinear system of codimension 1 defined by $\ker{\xi}$, where $\xi$ is considered as a linear functional on $\cH^0(\fO(D + K))$.
                    For any effective divisor $E$ of degree $d - 1$, let $L_E \subseteq |D + K|$ be the sublinear system $|D + K - E| + E$.
                    Show that $\fE$ is normalized if and only if for each $E$ as above, $L_E \nsubseteq H$.
              \item[(c)] Now show that if $-g \leq e < 0$, there exists a ruled surface $X$ over $C$ with invariant $e$.
              \item[(d)] For $g = 2$, show that $e \geq -2$ is also necessary for the existence of $X$.
          \end{itemize}

          \begin{proof} $ $ \vspace{0pt}
              \begin{itemize} [leftmargin=0cm]
                  \item[(a)] Let $\ge$ be a divisor of degree $-e$ on $C$, and let $\gk$ is a canonical divisor of $C$.
                        We need to first construct an indecomposable extension $\fE$ of $\fO(\ge)$ by $\fO$.
                        By (III, Ex. 6.1), an extension of vector bundles
                        \[ \begin{tikzcd}
                                0 \arrow[r] & \fO \arrow[r] & \fE \arrow[r] & \fO(\ge) \arrow[r] & 0
                            \end{tikzcd} \]
                        corresponds to an element in $\Ext^1(\fO(\ge), \fO)$.
                        In this case, $\Ext^1(\fO(\ge), \fO) \cong \cH^1(\fO(-\ge))$ (III, 6.3) and (III, 6.7).
                        This is dual to $\cH^0(\fO(\gk + \ge))$, so it is enough to find $\ge$ such that $\cH^0(\fO(\gk + \ge)) \neq 0$.
                        The image of the canonical morphism $C \to |\gk| \cong \P^{g - 1}$ is either a rational curve of degree $g - 1$, or an embedding of degree $2g - 2$, depending on whether or not $C$ is hyperelliptic.
                        If $-\ge$ is effective, then $\cH^0(\fO(\gk + \ge))$ is the space of all hyperplane sections in $\P^{g - 1}$ with support containing $\ge$.
                        In any case, a hyperplane section of $C$ in $\P^{g - 1}$ consists of $2g - 2$ points counted with multiplicity.
                        Hence, for each $0 \leq e \leq 2g - 2$, there is a divisor $\ge$ of degree $-e$ such that $\cH^0(\fO(\gk + \ge)) \neq 0$, so there exists indecomposable extensions of $\fO(\ge)$ by $\fO$.

                        Now, let $\fE$ be an extension of $\fO(\ge)$ defined by a nonzero element in $\Ext^1(\fO(\ge), \fO)$.
                        It remains to show $\fE$ is normalized and indecomposable.
                        Clearly $\cH^0(\fE) \neq0$ by construction.
                        If $\fM$ is any invertible sheaf, then we have an exact sequence
                        \[ \begin{tikzcd}
                                0 \arrow[r] & \fM \arrow[r] & \fE \otimes \fM \arrow[r] & \fO(\ge) \otimes \fM \arrow[r] & 0.
                            \end{tikzcd} \]
                        If $\deg \fM < 0$, then we have $\cH^0(\fM) = 0$ and $\cH^0(\fO(\ge) \otimes \fM) = 0$, and therefore also $\cH^0(\fE \otimes \fM) = 0$.
                        Furthermore, if $\fE$ is decomposable, then it would be $\fO \oplus \fL$ for some $\fL$, by (2.12).
                        But $\bigwedge^2 \fE \cong \fO_C(\ge)$, so $\fL \cong \fO(\ge)$, so in fact this extension would have to split, which it doesn't.
                        Hence, for each $0 \leq e \leq 2g - 2$, there is a ruled surface $X$ over $C$ with invariant $e$, corresponding to an indecomposable $\fE$.

                  \item[(b)] Let $\xi \in \cH^1(\fO(-D))$ be the (non-zero) element defining the extension $\fE$.
                        If $E$ is any effective divisor of degree $d - 1$ on $C$, then we have a commutative diagram
                        \[ \begin{tikzcd}
                                \cH^0(\fO) \arrow[r, "\delta"] \arrow[d, "\alpha"'] & \cH^1(\fO(-D))) \arrow[d, "\beta"'] \\
                                \cH^0(\fO(E)) \arrow[r, "\gamma"]    & \cH^1(\fO(E - D))
                            \end{tikzcd} \]
                        where $\delta(1) = \xi$, $\alpha(1) = t$, a non-zero section defining $E$, and $\beta$ is induced from the map $\fO \to \fO(E)$ corresponding to $t$.
                        On the other hand, $\beta$ is dual to the map
                        \begin{equation*}
                            \beta^\vee : \cH^0(\fO(D + K - E)) \to \cH^0(\fO(D + K))
                        \end{equation*}
                        also induced by $t$.
                        The subspace of $\cH^0(\fO(D+K))$ corresponding to $L_E$ is precisely the image of $\beta^\vee$, so $\beta(\xi) = 0$ if and only if $L_E \subseteq H$.

                        If $\fE$ is normalized, then $\gamma$ is injective for any effective divisor $E$ of degree $d - 1$.
                        By the commutative diagram, we must have $\beta(\xi) \neq 0$.
                        Hence, $L_E \nsubseteq H$.

                        Conversely, if $\fE$ is not normalized, then $\cH^0(\fE \otimes \fM) \neq 0$ for some invertible sheaf $\fM$ with $\deg{\fM} < 0$, and let $m = -\deg{\fM}$.
                        We have the exact sequence
                        \[ \begin{tikzcd}
                                0 \arrow[r] & \fM \arrow[r] & \fE \otimes \fM \arrow[r] & \fO(D) \otimes \fM \arrow[r] & 0
                            \end{tikzcd}. \]
                        If $m \geq d$, then we have $\cH^0(\fE \otimes \fM) = 0$ since $\deg{\fO(D) \otimes \fM} < 0$, and if $m = d$, then either $\cH^0(\fO(D) \otimes \fM) = 0$, or $\fM = \fO(-D)$, whence the coboundary map $\delta$ is injective, hence $\cH^0(\fE \otimes \fM) = 0$.
                        So we can assume $0 < m < d$.
                        The long exact sequence of cohomology gives
                        \[ \begin{tikzcd}
                                0 \arrow[r] & \cH^0(\fE \otimes \fM) \arrow[r] & \cH^0(\fO(D) \otimes \fM) \arrow[r] & \cH^1(\fM)
                            \end{tikzcd}, \]
                        where $\cH^0(\fM) = 0$.
                        The middle term is non-zero, so if $\fM \cong \fO(-M)$ with $m = \deg{M}$, then $D - M \sim E$ for some effective divisor $E$ of degree $d - m$.
                        For any effective divisor $\tilde{E}$ of degree $d - 1$ with $\tilde{E} - E \geq 0$, we have a commutative diagram,
                        \[ \begin{tikzcd}
                                \cH^0(\fO)) \arrow[r, "\delta"] \arrow[d, "\alpha"] \arrow[dd, "\tilde{\alpha}"', bend right=70] & \cH^1(\fO(-D)) \arrow[d, "\beta"] \arrow[dd, "\tilde{\beta}", bend left=70] \\
                                \cH^0(\fO(E)) \arrow[r, "\gamma"] \arrow[d, "\phi"]                                              & \cH^1(\fO(E - D)) \arrow[d, "\psi"]                                            \\
                                \cH^0(\fO(\tilde{E})) \arrow[r, "\eta"]                                                          & \cH^1(\fO(\tilde{E} - D))
                            \end{tikzcd}, \]
                        where $\delta(1) = \xi$, $\alpha, \beta$ is induced by a section $t \in \cH^0(\fO(E))$ defining $E$, and $\phi, \psi$ is induced by a section $s$ defining $\tilde{E} - E$.

                        Now, if $\beta(\xi) = 0$, then we are done.
                        Otherwise, dualizing the column on the right of the commutative diagram gives
                        \[ \begin{tikzcd}
                                \cH^0(\fO(D + K - \tilde{E})) \arrow[d, "\psi^\vee"] \arrow[rd, "\tilde{\beta}^\vee"] &                 \\
                                \cH^0(\fO(D + K - E)) \arrow[r, "\beta^\vee"]                        & \cH^0(\fO(D+K))
                            \end{tikzcd}. \]
                        Thus, for every $\tilde{E}$, we have $L_{\tilde{E}} \subseteq L_E$.
                        By the assumption $\beta(\xi) \neq 0$, we have $L_E \nsubseteq H$, so $\im(\beta^\vee) \cap \ker(\xi)$ is a nonempty proper subspace of $\im(\beta^\vee)$.
                        By the Riemann-Roch theorem, we have
                        \begin{equation*}
                            \dim \cH^0(\fO(D + K - \tilde{E})) = g, \quad \dim \cH^0(\fO(D + K - E)) = g + m - 1.
                        \end{equation*}
                        The space of possible $\tilde{E}$ has dimension $m - 1$, so the image of $\psi^\vee$ as $\tilde{E}$ varies is dense inside $\cH^0(\fO(D + K - E))$.
                        Consequently, we can find at least one $\tilde{E}$ such that $\im(\tilde{\beta}^\vee) \nsubseteq \ker(\xi)$, which is what we wanted to show.

                  \item[(c)] Let $\P^N = |K + D|$ with $N = g + d - 2$.
                        For any effective divisor $E$ of degree $d - 1$ on $C$, let us consider the set
                        \begin{equation*}
                            B_E = \{ \text{hyperplanes $H \subset \P^N$ | $L_E \subseteq H$} \} \subset |H|,
                        \end{equation*}
                        where $|H|$ is the complete linear system of hyperplanes in $\P^N$, i.e., $|H| = |K + D|$.
                        A hyperplane in $\P^N$ is determined by a nonzero global section $f \in V = \cH^0(\P^N, \fO_{\P^N}(1))$.
                        Viewing $L_E$ as a linear subspace of $\P^N$, let $W_E = \cH^0(L_E, \fO_{L_E}(1))$.
                        The space of effective divisors of degree $d - 1$ is parameterized by the closed points of the symmetric product $C^{(d - 1)}$, which has dimension $d - 1$.
                        Consider the restriction map
                        \begin{equation*}
                            \varphi_E : V \to W_E.
                        \end{equation*}
                        It is clearly surjective, and the hyperplanes $H \in B_E$ correspond exactly to those $f \in \ker \varphi_E$.
                        We have $\dim{V} = g + d - 1$ and $\dim{W_E} = g$, so $\dim{B_E} = \dim(\ker{\varphi_E}) - 1 = d - 2$.

                        Now, consider the subset $B \subset C^{(d - 1)} \times |H|$ consisting of all pairs $\langle E, H \rangle$ such that $E \in C^{(d - 1)}$ is a closed point and $H \in B_E$.
                        Clearly $B$ is the set of closed points of a $C^{(d - 1)} \times |H|$, which we denote also by $B$, and which we give a reduced induced scheme structure.
                        The first projection $p_1 : B \to C^{(d - 1)}$ is surjective, with fibre a projective space of dimension $d - 2$.
                        So $B$ has dimension $(d - 1) + (d - 2) = 2d - 3$, which is strictly smaller than $\dim |H| = g + d - 2$.
                        If $H \in |H| - p_2(B)$, then $H \nsupseteq L_E$ for every $E \in C^{(d - 1)}$, so the extension of $\fO(D)$ by $\fO$ given by the element in $\cH^0(\fO(K + D)) \cong \cH^1(\fO(-D))$ corresponding to $H$ satisfies the requirement of (b).
                        Hence, if $-g \leq e < 0$, then there exists a ruled surface $X$ over $C$ with invariant $e$.

                  \item[(d)] Let $C$ be a curve of genus $g = 2$, let $\fE$ be a normalized locally free sheaf of rank 2 on $C$ with invariant $e$, let $d = -e$, and let $X = \P(\fE)$ be the ruled surface associated to $\fE$.
                        As in the proof of (2.15), we want to show $e = -3, -4$ does not occur.

                        If it did, we would have a normalized bundle $\fE$ with an exact sequence
                        \[ \begin{tikzcd}
                                0 \arrow[r] & \fO \arrow[r] & \fE \arrow[r] & \fO(D) \arrow[r] & 0
                            \end{tikzcd}, \]
                        where $D$ is an effective divisor of degree $d = -e$.
                        Let $\xi \in \cH^1(\fO(-D))$ be the element defining the extension $\fE$.
                        If $E$ is any effective divisor, then applying $\Hom(\fO(D - E), -)$ and taking the long exact sequence in cohomology gives
                        \[ \begin{tikzcd}
                                \Hom(\fO(D - E), \fE) \arrow[r] & \Hom(\fO(D - E), \fO(D)) \arrow[r, "\delta_E"] & \Ext^1(\fO(D - E), \fO)
                            \end{tikzcd}, \]
                        so by exactness, the inclusion $i : \fO(D - E) \to \fO(D)$ factors through $\fE \to \fO(D)$ if and only if $\delta_E(i) = 0$.

                        By naturality, we have a commutative diagram
                        \[ \begin{tikzcd}
                                {\Hom(\fO(D), \fO(D))} \arrow[d, "i^*"] \arrow[r, "\delta"] & \Ext^1(\fO(D),  \fO) \arrow[d, "i^*"] \\
                                {\Hom(\fO(D - E), \fO(D))} \arrow[r, "\delta_E"]       & {\Ext^1(\fO(D - E), \fO))}
                            \end{tikzcd}, \]
                        where $\delta(1) = \xi$ and $i^*(1) = i$, so $\delta_E(i) = 0$ if and only if $i^*(\xi) = 0$.
                        Now, if there exists $E$ of degree $< d$ such that $i : \fO(D - E) \to \fE$ factors through $\fE \to \fO(D)$, then $\fO(D - E) \to \fE$ must also be injective, which is a contradiction of $\fE$ being normalized, so it is enough to show there exists $E$ such that $i^*(\xi) = 0$.

                        The discussion above has the following geometric interpretation.
                        By Serre duality, there is a natural identification
                        \begin{equation*}
                            |D + K| \cong \P(\Ext^1(\fO(D), \fO)) \cong \P(\cH^0(\fO(D + K))^\vee),
                        \end{equation*}
                        so we can view the linear span of $\xi$ in $\Ext^1(\fO(D), \fO)$ as a point in $|D + K|$, which we will also denote by $\xi$.
                        It follows that $i^*(\xi) = 0$ if and only if $\xi$ is contained in every hyperplane in $|D + K|$ which also contains $E$.

                        \fbox{$d = 3$} In either case, $D + K$ is a very ample divisor on $C$ (IV, Ex. 3.1), so consider the embedding of $C$ in $|D + K| \cong \P^3$.
                        The secant variety of $C$ fills up the entire space (IV, \SS 3), so there is a secant line $L \subseteq |D + K|$ which contains $\xi$.
                        A line is uniquely determined by two points, so we can take $E$ to be any effective divisor of degree 2 with support contained in $C \cap L$.
                        Hence, any hyperplane $H \subseteq |D + K|$ containing $E$ also contains $L$, so $\xi$ must be a point in $H$ as well.

                        \fbox{$d = 4$} Again, consider $C$ in its embedding in $|D + K| \cong \P^4$.
                        A plane is uniquely determined by three points, and the union of all planes going through at least three points of $C$ fills up all of $|D + K|$.
                        This means we can find a plane $P$ in $|D + K|$ which contains $\xi$ and at least three points of $C$.
                        So we can take $E$ to be any effective divisor of degree 3 with support contained in $C \cap P$.
                        Hence, any hyperplane $H \subseteq |D + K|$ containing $E$ also contains $P$, so $\xi$ must be in $H$ as well.
              \end{itemize}
          \end{proof}

    \item Show that every locally free sheaf of finite rank on $\P^1$ is isomorphic to a direct sum of invertible sheaves.
          \filbreak

          \begin{proof}
              We call any locally free sheaf $\fE$ on a curve $C$ \textit{normalized} if $\cH^0(C, \fE) \neq0$ and $\cH^0(C, \fE \otimes \fL) = 0$ for all invertible sheaves $\fL$ on $C$ with $\deg{\fL} < 0$.
              With this definition, the statement of (2.8) can be made sense for $\fE$ of arbitrary rank:

              \begin{lemma}
                  For every locally free sheaf $\fE$ of finite rank on a curve $C$, there exists an invertible sheaf $\fL$ on $C$ such that $\fE \otimes \fL$ is normalized.
              \end{lemma}

              Going back to the original statement, we can assume $\fE$ is normalized.
              We proceed by induction on the rank $r$ of $\fE$.
              The base case $r = 1$ is trivial, so assume the statement holds for any $< r$.
              Pick a subinvertible sheaf $0 \to \fO \to \fE$ of maximal degree, and consider the short exact sequence
              \[ \begin{tikzcd}
                      0 \arrow[r] & \fO \arrow[r] & \fE \arrow[r] & \fF \arrow[r] & 0
                  \end{tikzcd}. \]
              The quotient sheaf $\fF = \fE / \fO$ is localy free of rank $r - 1$ because of the choice of $0 \to \fO \to \fE$ (2.8), so by the inductive hypothesis, $\fF$ is decomposable, say $\fF = \bigoplus \fO(b_i)$ for some $b_i \in \Z$.
              Tensoring by $\fO(-1)$ and taking cohomology gives
              \[ \begin{tikzcd}
                      0 \arrow[r] & \cH^0(\fO(-1)) \arrow[r] & \cH^0(\fE(-1)) \arrow[r] & \cH^0(\bigoplus \fO(b_i - 1)) \arrow[r] & 0
                  \end{tikzcd}. \]
              Since $\fE$ is normalized, the middle vector space is zero, so by exactness we have $b_i \leq 0$.
              On the other hand, the space of extensions of $\fF$ by $\fO$ can be computed by Serre duality, which gives
              \begin{equation*}
                  \Ext^1(\fF, \fO) \cong \cH^0(\bigoplus \fO(b_i - 1)) = 0,
              \end{equation*}
              so any extension of $\fF$ by $\fO$ is trivial.
              Hence, the short exact sequence splits, and there is an isomorphism $\fE \cong \fO \oplus \fF$.
          \end{proof}

                            \alpha'(s_2) & = u \dd(tu^2) - t \dd(u^3) = u^2(u \dd t - t \dd u),
                        \end{align*}
                        so $\alpha$ is an isomorphism.
                        Hence, we have
                        \begin{equation*}
                            \fN_{C/\P^3} \cong \fO_C(5) \oplus \fO_C(5).
                        \end{equation*}

                  \item[(b)] We can do the same calculation as above, except in characteristic 2, the corresponding $\alpha'$ has a nontrivial kernel of rank 1.
                        Explicitly, we have
                        \begin{equation*}
                            \alpha' : \cH^0(\restr{\Omega_{\P^3}}{C}(6)) \to \cH^0(\Omega_C(6)),
                        \end{equation*}
                        where we can identify $\Omega_C(6) \cong \fO_C(4)$, and
                        \begin{equation*}
                            \cH^0(\restr{\Omega_{\P^3}}{C}(6)) = \ker(\beta' : \cH^0(\fO_C(2)^{\oplus 4}) \to \cH^0(\fO_C(6))).
                        \end{equation*}
                        Consider the section of $\restr{\Omega_{\P^3}}{C}(6)$ defined by
                        \begin{equation*}
                            s = u^2 \dd x_1 - t^2 \dd x_2.
                        \end{equation*}
                        Its image under $\beta$ can be computed by plugging $x_2 = t^3 u$ and $x_3 = tu^3$,
                        \begin{align*}
                            \beta'(s) & = u^2 \dd(t^3 u) - t^2 \dd (tu^3)                               \\
                                      & = u^2(3t^2 u \dd t + t^3 \dd u) - t^2(u^3 \dd t + 3 tu^2 \dd u) \\
                                      & = 2 t^2 u^2 (u \dd t - t \dd u),
                        \end{align*}
                        which is zero if $\car{k} = 2$.
                        To summarize,
                        \begin{equation*}
                            \fN_{C/\P^3} = \begin{cases*}
                                \fO_C(6) \oplus \fO_C(8) & $\car{k} = 2$, \\
                                \fO_C(7) \oplus \fO_C(7) & otherwise.
                            \end{cases*}
                        \end{equation*}
              \end{itemize}
          \end{proof}

\end{enumerate}
